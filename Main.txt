local function loadScript(name, url)
    print("[Loader] กำลังโหลด " .. name .. "...")

    local success, result = pcall(function()
        local source = game:HttpGet(url)

        if not source or source == "" then
            error("โหลดไฟล์ว่าง")
        end

        local func, compileError = loadstring(source)

        if not func then
            error(compileError)
        end

        return func
    end)

    if not success then
        warn("[Loader] " .. name .. " โหลดไม่สำเร็จ:", result)
        return nil
    end

    print("[Loader] " .. name .. " โหลดสำเร็จ")
    return result
end

-- ==================================================
-- Anti-AFK
-- ==================================================

if getgenv().AntiAFKStarted then

    print("[Loader] Anti-AFK ทำงานอยู่แล้ว -> ข้าม")

else

    if not getgenv().AntiAFKLoaded then

        local AntiAFK = loadScript(
            "Anti-AFK",
            "https://raw.githubusercontent.com/r6dt/all/refs/heads/main/antiafk.txt"
        )

        if not AntiAFK then
            warn("[Loader] Anti-AFK โหลดไม่ได้ -> หยุด")
            return
        end

        getgenv().AntiAFKLoaded = true

        task.spawn(function()
            local success, err = pcall(AntiAFK)

            if not success then
                warn("[AntiAFK] ERROR:", err)
                getgenv().AntiAFKLoaded = nil
                getgenv().AntiAFKStarted = nil
            end
        end)

    else
        print("[Loader] Anti-AFK กำลังเริ่มทำงาน...")
    end

    -- รอ Anti-AFK แจ้งว่าเริ่มทำงานจริง
    print("[Loader] กำลังตรวจสอบ Anti-AFK...")

    local timeout = 15
    local elapsed = 0

    while not getgenv().AntiAFKStarted and elapsed < timeout do
        task.wait(0.5)
        elapsed += 0.5
    end

    if not getgenv().AntiAFKStarted then
        warn("[Loader] Anti-AFK ไม่เริ่มทำงานภายใน 15 วินาที")
        warn("[Loader] ยกเลิกการรัน ForFarm")
        return
    end

    print("[Loader] Anti-AFK เริ่มทำงานสำเร็จ!")

end
