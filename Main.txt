repeat task.wait() until game:IsLoaded()
task.wait(10)

print("[Loader] เริ่มทำงาน")

local ANTI_AFK_URL =
    "https://raw.githubusercontent.com/r6dt/all/refs/heads/main/antiafk.txt"

local FOR_FARM_URL =
    "https://raw.githubusercontent.com/r6dt/all/refs/heads/main/ForFarm.txt"

local MAX_RETRY = 3
local START_TIMEOUT = 15

-- รอ 300 วินาที ก่อนลอง AntiAFK ใหม่
local ANTI_AFK_RETRY_INTERVAL = 300

local WAIT_BEFORE_FARM = 5

-- ==================================================
-- LOAD SCRIPT
-- ==================================================

local function loadScript(name, url)

    print("[Loader] กำลังโหลด " .. name .. "...")

    local success, result = pcall(function()

        local source = game:HttpGet(url)

        if not source or source == "" then
            error("โหลดไฟล์ว่าง")
        end

        local func, compileError = loadstring(source)

        if not func then
            error(compileError)
        end

        return func
    end)

    if not success then

        warn(
            "[Loader] " .. name .. " โหลดไม่สำเร็จ:",
            result
        )

        return nil
    end

    print("[Loader] " .. name .. " โหลดสำเร็จ")

    return result
end

-- ==================================================
-- START ANTI-AFK
-- ==================================================

local function startAntiAFK()

    -- ถ้าทำงานอยู่แล้ว ไม่ต้องโหลดซ้ำ
    if getgenv().AntiAFKStarted == true then

        print(
            "[Loader] Anti-AFK ทำงานอยู่แล้ว -> ข้าม"
        )

        return true
    end

    print("[Loader] กำลังเริ่ม Anti-AFK...")

    getgenv().AntiAFKStarted = nil
    getgenv().AntiAFKLoaded = nil

    local AntiAFK = loadScript(
        "Anti-AFK",
        ANTI_AFK_URL
    )

    if not AntiAFK then

        warn(
            "[Loader] โหลด Anti-AFK ไม่สำเร็จ"
        )

        return false
    end

    task.spawn(function()

        local success, err = pcall(AntiAFK)

        if not success then

            warn(
                "[AntiAFK] ERROR:",
                err
            )

            -- ถ้า Script Error ให้ถือว่ายังไม่ทำงาน
            getgenv().AntiAFKStarted = nil
        end
    end)

    print(
        "[Loader] รอ Anti-AFK เริ่มทำงาน..."
    )

    local elapsed = 0

    while not getgenv().AntiAFKStarted
        and elapsed < START_TIMEOUT do

        task.wait(0.5)
        elapsed += 0.5

    end

    if getgenv().AntiAFKStarted == true then

        print(
            "[Loader] ✅ Anti-AFK เริ่มทำงานสำเร็จ"
        )

        getgenv().AntiAFKLoaded = true

        return true

    end

    warn(
        "[Loader] ❌ Anti-AFK ไม่เริ่มทำงานภายใน "
        .. START_TIMEOUT
        .. " วินาที"
    )

    return false
end

-- ==================================================
-- ANTI-AFK INITIAL START
-- ==================================================

local antiAFKSuccess = false

for attempt = 1, MAX_RETRY do

    print(
        "[Loader] Anti-AFK Attempt "
        .. attempt
        .. "/"
        .. MAX_RETRY
    )

    if startAntiAFK() then

        antiAFKSuccess = true
        break

    end

    if attempt < MAX_RETRY then

        print(
            "[Loader] รอ 2 วินาทีก่อน Retry..."
        )

        task.wait(2)
    end
end

-- ==================================================
-- ANTI-AFK WATCHDOG
-- ==================================================

task.spawn(function()

    while true do

        task.wait(ANTI_AFK_RETRY_INTERVAL)

        -- ถ้า AntiAFK ยังทำงานอยู่
        if getgenv().AntiAFKStarted == true then

            print(
                "[Loader] Anti-AFK STATUS: RUNNING"
            )

        else

            warn(
                "[Loader] ⚠️ Anti-AFK ไม่ทำงาน"
            )

            warn(
                "[Loader] 🔄 กำลังลอง Anti-AFK ใหม่..."
            )

            if startAntiAFK() then

                print(
                    "[Loader] ✅ Anti-AFK กลับมาทำงานแล้ว"
                )

                antiAFKSuccess = true

            else

                warn(
                    "[Loader] ❌ Anti-AFK Retry ไม่สำเร็จ"
                )
            end
        end
    end
end)

-- ==================================================
-- ANTI-AFK FAILED
-- ==================================================

if not antiAFKSuccess then

    warn(
        "[Loader] ⚠️ Anti-AFK เริ่มต้นไม่สำเร็จ"
    )

    warn(
        "[Loader] แต่ Watchdog จะลองใหม่ทุก "
        .. ANTI_AFK_RETRY_INTERVAL
        .. " วินาที"
    )

else

    print(
        "[Loader] ✅ Anti-AFK พร้อมทำงาน"
    )
end

-- ==================================================
-- WAIT BEFORE FORFARM
-- ==================================================

print(
    "[Loader] รอ "
    .. WAIT_BEFORE_FARM
    .. " วินาที ก่อนเริ่ม ForFarm..."
)

task.wait(WAIT_BEFORE_FARM)

-- ==================================================
-- FORFARM
-- ==================================================

if getgenv().ForFarmLoaded then

    print(
        "[Loader] ForFarm รันอยู่แล้ว -> ข้าม"
    )

else

    print(
        "[Loader] กำลังโหลด ForFarm..."
    )

    local ForFarm = loadScript(
        "ForFarm",
        FOR_FARM_URL
    )

    if not ForFarm then

        warn(
            "[Loader] ForFarm โหลดไม่สำเร็จ"
        )

        return
    end

    getgenv().ForFarmLoaded = true

    task.spawn(function()

        local success, err = pcall(ForFarm)

        if not success then

            warn(
                "[ForFarm] ERROR:",
                err
            )

            getgenv().ForFarmLoaded = nil
        end
    end)

    print(
        "[Loader] ✅ ForFarm เริ่มทำงานแล้ว!"
    )
end

print("[Loader] =================================")
print("[Loader] โหลดทุกอย่างเสร็จแล้ว")
print("[Loader] Anti-AFK Watchdog: 300 วินาที")
print("[Loader] =================================")
