print("[Loader] เริ่มทำงาน")

local ANTI_AFK_URL =
    "https://raw.githubusercontent.com/r6dt/all/refs/heads/main/antiafk.txt"

local FOR_FARM_URL =
    "https://raw.githubusercontent.com/r6dt/all/refs/heads/main/ForFarm.txt"

local MAX_RETRY = 3
local START_TIMEOUT = 15
local WAIT_BEFORE_FARM = 5

-- ==================================================
-- LOAD SCRIPT
-- ==================================================

local function loadScript(name, url)
    print("[Loader] กำลังโหลด " .. name .. "...")

    local success, result = pcall(function()

        local source = game:HttpGet(url)

        if not source or source == "" then
            error("โหลดไฟล์ว่าง")
        end

        local func, compileError = loadstring(source)

        if not func then
            error(compileError)
        end

        return func
    end)

    if not success then
        warn("[Loader] " .. name .. " โหลดไม่สำเร็จ:", result)
        return nil
    end

    print("[Loader] " .. name .. " โหลดสำเร็จ")

    return result
end

-- ==================================================
-- ANTI-AFK
-- ==================================================

local antiAFKSuccess = false

if getgenv().AntiAFKStarted then

    print("[Loader] Anti-AFK ทำงานอยู่แล้ว -> ข้าม")
    antiAFKSuccess = true

else

    for attempt = 1, MAX_RETRY do

        print(
            "[Loader] Anti-AFK Attempt "
            .. attempt
            .. "/"
            .. MAX_RETRY
        )

        -- ล้างสถานะก่อนลองใหม่
        getgenv().AntiAFKStarted = nil

        local AntiAFK = loadScript(
            "Anti-AFK",
            ANTI_AFK_URL
        )

        if AntiAFK then

            task.spawn(function()

                local success, err = pcall(AntiAFK)

                if not success then
                    warn("[AntiAFK] ERROR:", err)
                end

            end)

            print("[Loader] รอ Anti-AFK เริ่มทำงาน...")

            local elapsed = 0

            while not getgenv().AntiAFKStarted
                and elapsed < START_TIMEOUT do

                task.wait(0.5)
                elapsed += 0.5

            end

            if getgenv().AntiAFKStarted then

                print(
                    "[Loader] Anti-AFK สำเร็จ! "
                    .. "Attempt "
                    .. attempt
                )

                getgenv().AntiAFKLoaded = true
                antiAFKSuccess = true

                break

            else

                warn(
                    "[Loader] Anti-AFK ไม่เริ่มทำงาน "
                    .. "ภายใน "
                    .. START_TIMEOUT
                    .. " วินาที"
                )

            end

        end

        if attempt < MAX_RETRY then
            print("[Loader] รอ 2 วินาทีก่อน Retry...")
            task.wait(2)
        end
    end
end

-- ==================================================
-- ANTI-AFK FAILED
-- ==================================================

if not antiAFKSuccess then

    warn(
        "[Loader] Anti-AFK ล้มเหลวครบ "
        .. MAX_RETRY
        .. " ครั้ง"
    )

    warn("[Loader] จะไม่รัน ForFarm")

    return
end

-- ==================================================
-- WAIT BEFORE FORFARM
-- ==================================================

print(
    "[Loader] Anti-AFK OK -> "
    .. "รอ "
    .. WAIT_BEFORE_FARM
    .. " วินาที..."
)

task.wait(WAIT_BEFORE_FARM)

-- ==================================================
-- FORFARM
-- ==================================================

if getgenv().ForFarmLoaded then

    print("[Loader] ForFarm รันอยู่แล้ว -> ข้าม")

else

    print("[Loader] กำลังโหลด ForFarm...")

    local ForFarm = loadScript(
        "ForFarm",
        FOR_FARM_URL
    )

    if not ForFarm then

        warn("[Loader] ForFarm โหลดไม่สำเร็จ")

        return
    end

    getgenv().ForFarmLoaded = true

    task.spawn(function()

        local success, err = pcall(ForFarm)

        if not success then

            warn("[ForFarm] ERROR:", err)

            getgenv().ForFarmLoaded = nil

        end

    end)

    print("[Loader] ForFarm เริ่มทำงานแล้ว!")

end

print("[Loader] =================================")
print("[Loader] โหลดทุกอย่างเสร็จแล้ว")
print("[Loader] =================================")
