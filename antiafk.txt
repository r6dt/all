end

local blockedJobs = getgenv().AntiAFKBlockedJobs

-- =========================================================
-- LOAD JOB FILE
-- =========================================================

local function loadJobFile()

    if not hasFileAPI then
        return
    end

    if not isfile(JOB_FILE) then
        print("[AntiAFK] ยังไม่มีไฟล์", JOB_FILE)
        return
    end

    local success, content = pcall(function()
        return readfile(JOB_FILE)
    end)

    if not success then
        warn("[AntiAFK] อ่านไฟล์ JobId ไม่สำเร็จ:", content)
        return
    end

    if not content or content == "" then
        print("[AntiAFK] ไฟล์ JobId ว่าง")
        return
    end

    local count = 0

    for line in string.gmatch(content, "[^\r\n]+") do

        local jobId = line:match("^([^|]+)")

        if jobId and jobId ~= "" then
            blockedJobs[jobId] = true
            count += 1
        end
    end

    print("[AntiAFK] โหลด JobId จากไฟล์แล้ว:", count, "รายการ")
end

loadJobFile()

-- =========================================================
-- SAVE JOB FILE
-- =========================================================

local function saveJobFile()

    if not hasFileAPI then
        return false
    end

    local jobs = {}

    for jobId in pairs(blockedJobs) do
        table.insert(jobs, jobId)
    end

    -- จำกัดจำนวน
    while #jobs > MAX_BLOCKED_JOBS do
        table.remove(jobs, 1)
    end

    local output = ""

    for _, jobId in ipairs(jobs) do
        output = output .. jobId .. "\n"
    end

    local success, err = pcall(function()
        writefile(JOB_FILE, output)
    end)

    if not success then
        warn("[AntiAFK] บันทึก JobId ไม่สำเร็จ:", err)
        return false
    end

    print(
        "[AntiAFK] บันทึก JobId ลง",
        JOB_FILE,
        "แล้ว"
    )

    return true
end

-- =========================================================
-- ADD BLOCKED JOB
-- =========================================================

local function addBlockedJob(jobId)

    if not jobId or jobId == "" then
        return
    end

    if blockedJobs[jobId] then
        print("[AntiAFK] JobId นี้มีอยู่แล้ว:", jobId)
        return
    end

    blockedJobs[jobId] = true

    print("================================")
    print("[AntiAFK] เพิ่ม JobId ที่ถูก Kick")
    print("[AntiAFK] JobId:", jobId)
    print("================================")

    -- จำกัดจำนวน JobId
    local jobs = {}

    for id in pairs(blockedJobs) do
        table.insert(jobs, id)
    end

    while #jobs > MAX_BLOCKED_JOBS do

        local removeId = table.remove(jobs, 1)

        if removeId then
            blockedJobs[removeId] = nil
        end
    end

    saveJobFile()
end

-- =========================================================
-- CHECK BLOCKED JOB
-- =========================================================

local function isBlockedJob(jobId)

    if not jobId or jobId == "" then
        return false
    end

    return blockedJobs[jobId] == true
end

-- =========================================================
-- PRINT JOB LIST
-- =========================================================

local function printBlockedJobs()

    local count = 0

    print("[AntiAFK] ===== Blocked JobIds =====")

    for jobId in pairs(blockedJobs) do
        count += 1

        print(
            "[AntiAFK]",
            count .. ".",
            jobId
        )
    end

    print(
        "[AntiAFK] ทั้งหมด:",
        count,
        "JobId"
    )

    print("[AntiAFK] =========================")
end

printBlockedJobs()

local samePositionCount = 0
local lastPosition = nil

local function getPosition()

    local character = Player.Character

    if not character then
        return nil
    end

    local root =
        character:FindFirstChild("HumanoidRootPart")

    if not root then
        return nil
    end

    return root.Position
end

-- =========================================================
-- POSITION CHECK
-- =========================================================

local function isSamePosition(pos1, pos2)

    if not pos1 or not pos2 then
        return false
    end

    local distance =
        (pos1 - pos2).Magnitude

    print(
        "[AntiAFK] ระยะที่ขยับ:",
        math.floor(distance * 100) / 100,
        "studs"
    )

    return distance <= POSITION_TOLERANCE
end

-- =========================================================
-- QUEUE SCRIPT AFTER TELEPORT
-- =========================================================

local function queueAfterTeleport()

    local queue =
        queue_on_teleport
        or queueonteleport

    if not queue then

        warn(
            "[ServerHop] Executor ไม่มี queue_on_teleport"
        )

        return false
    end

    local success, err = pcall(function()

        queue([[
            task.wait(3)

            local success, err = pcall(function()

                loadstring(game:HttpGet(
                    "https://raw.githubusercontent.com/r6dt/all/refs/heads/main/antiafk.txt"
                ))()

            end)

            if not success then
                warn("[AntiAFK Queue] ERROR:", err)
            end
        ]])

    end)

    if not success then

        warn(
            "[ServerHop] Queue ไม่สำเร็จ:",
            err
        )

        return false
    end

    print("[ServerHop] Queue AntiAFK สำหรับ Server ใหม่แล้ว")

    return true
end

-- =========================================================
-- SERVER HOP
-- =========================================================

local function serverHop()

    print("================================")
    print("[ServerHop] เริ่มหา Server ใหม่")
    print("[ServerHop] Current JobId:", game.JobId)
    print("================================")

    local placeId = game.PlaceId

    -- -----------------------------------------------
    -- QUEUE
    -- -----------------------------------------------

    if not queueAfterTeleport() then
        warn("[ServerHop] Queue ไม่สำเร็จ")
    end

    -- -----------------------------------------------
    -- GET SERVER LIST
    -- -----------------------------------------------

    local success, response = pcall(function()

        return game:HttpGet(
            "https://games.roblox.com/v1/games/"
            .. placeId
            .. "/servers/Public?sortOrder=Asc&limit=100"
        )

    end)

    if not success then

        warn(
            "[ServerHop] ดึง Server List ไม่สำเร็จ:",
            response
        )

        return false
    end

    -- -----------------------------------------------
    -- DECODE
    -- -----------------------------------------------

    local decodeSuccess, data = pcall(function()

        return HttpService:JSONDecode(response)

    end)

    if not decodeSuccess
        or not data
        or not data.data then

        warn(
            "[ServerHop] อ่าน Server List ไม่สำเร็จ"
        )

        return false
    end

    -- -----------------------------------------------
    -- FIND SERVER
    -- -----------------------------------------------

    for _, server in ipairs(data.data) do

        if server.id ~= game.JobId
            and server.playing < server.maxPlayers then

            if not isBlockedJob(server.id) then

                print(
                    "[ServerHop] พบ Server ใหม่:",
                    server.id
                )

                print(
                    "[ServerHop] Players:",
                    server.playing
                    .. "/"
                    .. server.maxPlayers
                )

                local teleportSuccess,
                    teleportError =
                    pcall(function()

                        TeleportService:
                            TeleportToPlaceInstance(
                                placeId,
                                server.id,
                                Player
                            )

                    end)

                if teleportSuccess then

                    print(
                        "[ServerHop] กำลัง Teleport..."
                    )

                    return true

                else

                    warn(
                        "[ServerHop] Teleport ไม่สำเร็จ:",
                        teleportError
                    )

                end

            else

                print(
                    "[ServerHop] ข้าม JobId ที่ถูกบันทึก:",
                    server.id
                )

            end
        end
    end

    warn(
        "[ServerHop] ไม่พบ Server ใหม่"
    )

    return false
end

-- =========================================================
-- MAIN LOOP
-- =========================================================

print("[AntiAFK] Main Loop เริ่มทำงานแล้ว")

while true do

    task.wait(CHECK_INTERVAL)

    local currentPosition =
        getPosition()

    if not currentPosition then

        warn(
            "[AntiAFK] ไม่พบ Character/HumanoidRootPart"
        )

        continue
    end

    -- -----------------------------------------------
    -- FIRST CHECK
    -- -----------------------------------------------

    if not lastPosition then

        lastPosition = currentPosition
        samePositionCount = 0

        print(
            "[AntiAFK] บันทึกตำแหน่งเริ่มต้นแล้ว"
        )

        print(
            "[AntiAFK] Check: 0/"
            .. REQUIRED_CHECKS
        )

    -- -----------------------------------------------
    -- SAME POSITION
    -- -----------------------------------------------

    elseif isSamePosition(
        lastPosition,
        currentPosition
    ) then

        samePositionCount += 1

        print(
            "[AntiAFK] อยู่จุดเดิม | "
            .. samePositionCount
            .. "/"
            .. REQUIRED_CHECKS
        )

        -- -------------------------------------------
        -- AFK TRIGGER
        -- -------------------------------------------

        if samePositionCount >= REQUIRED_CHECKS then

            print("================================")
            print(
                "[AntiAFK] ⚠️ AFK TRIGGERED"
            )
            print(
                "[AntiAFK] นิ่งครบ "
                .. REQUIRED_CHECKS
                .. " ครั้ง"
            )

            print(
                "[AntiAFK] กำลังบันทึก JobId..."
            )

            -- สำคัญ:
            -- บันทึก JobId ก่อนออกจาก Server
            addBlockedJob(game.JobId)

            print(
                "[AntiAFK] JobId ถูกบันทึกแล้ว"
            )

            print(
                "[ServerHop] กำลังหา Server ใหม่..."
            )

            print("================================")

            local hopped =
                serverHop()

            if hopped then

                print(
                    "[ServerHop] Teleport สำเร็จ"
                )

                getgenv().AntiAFKRunning = false

                break

            else

                warn(
                    "[ServerHop] Hop ไม่สำเร็จ -> Reset Check"
                )

                samePositionCount = 0
                lastPosition = currentPosition
            end
        end

    -- -----------------------------------------------
    -- POSITION CHANGED
    -- -----------------------------------------------

    else

        samePositionCount = 0

        print(
            "[AntiAFK] ตำแหน่งเปลี่ยน"
            .. " -> Reset 0/"
            .. REQUIRED_CHECKS
        )

    end

    lastPosition = currentPosition
end
