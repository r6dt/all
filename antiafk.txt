-- =========================================================
-- LOAD JOB FILE
-- =========================================================

local function loadJobFile()

    if not hasFileAPI then
        return
    end

    if not isfile(JOB_FILE) then
        print("[AntiAFK] ยังไม่มีไฟล์", JOB_FILE)
        return
    end

    local success, content = pcall(function()
        return readfile(JOB_FILE)
    end)

    if not success then
        warn("[AntiAFK] อ่าน Job File ไม่สำเร็จ:", content)
        return
    end

    if not content or content == "" then
        return
    end

    local count = 0

    for line in string.gmatch(content, "[^\r\n]+") do

        local jobId = line:match("^([^|]+)")

        if jobId then
            jobId = jobId:match("^%s*(.-)%s*$")
        end

        if jobId and jobId ~= "" then

            if not blockedJobs[jobId] then
                blockedJobs[jobId] = true
                count += 1
            end

        end
    end

    print("[AntiAFK] โหลด Blocked JobId:", count)
end

loadJobFile()

-- =========================================================
-- CHECK BLOCKED JOB
-- =========================================================

local function isBlockedJob(jobId)

    if not jobId or jobId == "" then
        return false
    end

    return blockedJobs[jobId] == true
end

-- =========================================================
-- COUNT BLOCKED
-- =========================================================

local function getBlockedJobCount()

    local count = 0

    for _ in pairs(blockedJobs) do
        count += 1
    end

    return count
end

-- =========================================================
-- SAVE JOB FILE
-- =========================================================

local function saveJobFile()

    if not hasFileAPI then
        return false
    end

    local jobs = {}

    for jobId in pairs(blockedJobs) do
        table.insert(jobs, jobId)
    end

    while #jobs > MAX_BLOCKED_JOBS do

        local removeId = table.remove(jobs, 1)

        if removeId then
            blockedJobs[removeId] = nil
        end
    end

    local output = ""

    for _, jobId in ipairs(jobs) do
        output = output .. jobId .. "\n"
    end

    local success, err = pcall(function()
        writefile(JOB_FILE, output)
    end)

    if not success then
        warn("[AntiAFK] บันทึก JobId ไม่สำเร็จ:", err)
        return false
    end

    return true
end

-- =========================================================
-- LOAD SERVER CACHE
-- =========================================================

local function loadServerCache()

    serverCache = {}

    if not hasFileAPI then
        warn("[AntiAFK] File API ไม่มี -> อ่าน server_cache.txt ไม่ได้")
        return false
    end

    if not isfile(SERVER_CACHE_FILE) then
        warn("[AntiAFK] ไม่พบ", SERVER_CACHE_FILE)
        return false
    end

    local success, content = pcall(function()
        return readfile(SERVER_CACHE_FILE)
    end)

    if not success then
        warn("[AntiAFK] อ่าน Server Cache ไม่สำเร็จ:", content)
        return false
    end

    if not content or content == "" then
        warn("[AntiAFK] server_cache.txt ว่าง")
        return false
    end

    local count = 0
    local duplicateCount = 0
    local blockedCount = 0

    for line in string.gmatch(content, "[^\r\n]+") do

        -- รองรับ:
        -- JobId
        -- JobId|จำนวนคน

        local jobId, playerCount =
            line:match("^([^|]+)|?(%d*)")

        if jobId then
            jobId = jobId:match("^%s*(.-)%s*$")
        end

        playerCount = tonumber(playerCount) or 0

        if jobId and jobId ~= "" then

            -- =================================================
            -- BLOCKED JOB
            -- =================================================

            if isBlockedJob(jobId) then

                blockedCount += 1

            -- =================================================
            -- DUPLICATE JOB
            -- =================================================

            elseif serverCache[jobId] ~= nil then

                duplicateCount += 1

                -- ถ้ามีซ้ำ ให้ใช้จำนวนคนล่าสุด
                serverCache[jobId] = playerCount

            -- =================================================
            -- NEW JOB
            -- =================================================

            else

                serverCache[jobId] = playerCount
                count += 1

            end
        end
    end

    print(
        "[AntiAFK] 📦 โหลด Server Cache:",
        count,
        "JobId"
    )

    if duplicateCount > 0 then
        print(
            "[AntiAFK] ♻️ กรอง JobId ซ้ำ:",
            duplicateCount
        )
    end

    if blockedCount > 0 then
        print(
            "[AntiAFK] 🚫 กรอง Blocked JobId:",
            blockedCount
        )
    end

    return count > 0
end

-- =========================================================
-- SAVE SERVER CACHE
-- =========================================================

local function saveServerCache()

    if not hasFileAPI then
        return false
    end

    local output = ""

    local count = 0
    local duplicateGuard = {}

    for jobId, playerCount in pairs(serverCache) do

        -- =================================================
        -- กันซ้ำอีกชั้น
        -- =================================================

        if not duplicateGuard[jobId] then

            duplicateGuard[jobId] = true

            -- ห้ามบันทึก Blocked Job
            if not isBlockedJob(jobId) then

                output =
                    output
                    .. jobId
                    .. "|"
                    .. tostring(playerCount or 0)
                    .. "\n"

                count += 1
            end
        end
    end

    local success, err = pcall(function()
        writefile(
            SERVER_CACHE_FILE,
            output
        )
    end)

    if not success then

        warn(
            "[AntiAFK] ❌ บันทึก Server Cache ไม่สำเร็จ:",
            err
        )

        return false
    end

    print(
        "[AntiAFK] 💾 อัปเดต Server Cache:",
        count,
        "JobId"
    )

    return true
end

-- =========================================================
-- UPDATE CURRENT SERVER
-- =========================================================

local function updateCurrentServerCache()

    local currentJobId = game.JobId

    if not currentJobId or currentJobId == "" then
        return false
    end

    -- =====================================================
    -- ถ้า Job ปัจจุบันถูก Block แล้ว
    -- ห้ามเพิ่มลง Cache
    -- =====================================================

    if isBlockedJob(currentJobId) then

        if serverCache[currentJobId] ~= nil then

            serverCache[currentJobId] = nil

            print(
                "[AntiAFK] 🚫 Current Job ถูก Block -> ลบออกจาก Cache:",
                currentJobId
            )

            saveServerCache()
        end

        return false
    end

    -- =====================================================
    -- จำนวนผู้เล่นปัจจุบัน
    -- =====================================================

    local playerCount =
        #Players:GetPlayers()

    -- =====================================================
    -- UPDATE
    -- =====================================================

    local existed =
        serverCache[currentJobId] ~= nil

    serverCache[currentJobId] =
        playerCount

    if existed then

        print(
            "[AntiAFK] 🔄 อัปเดต JobId เดิม:",
            currentJobId,
            "| ผู้เล่น:",
            playerCount
        )

    else

        print(
            "[AntiAFK] ➕ เพิ่ม JobId ใหม่:",
            currentJobId,
            "| ผู้เล่น:",
            playerCount
        )
    end

    saveServerCache()

    return true
end

-- =========================================================
-- REMOVE BLOCKED JOB FROM CACHE
-- =========================================================

local function removeBlockedJobsFromCache()

    local removed = 0

    for jobId in pairs(serverCache) do

        if isBlockedJob(jobId) then

            serverCache[jobId] = nil
            removed += 1

        end
    end

    if removed > 0 then

        print(
            "[AntiAFK] 🧹 ลบ Blocked Job จาก Cache:",
            removed
        )

        saveServerCache()
    end
end

-- =========================================================
-- CACHE COUNT
-- =========================================================

local function getServerCacheCount()

    local count = 0

    for _ in pairs(serverCache) do
        count += 1
    end

    return count
end

-- =========================================================
-- GET VALID CACHED JOBS
-- =========================================================

local function getValidCachedJobs()

    local currentJobId = game.JobId
    local maxPlayers = Players.MaxPlayers or math.huge

    local validJobs = {}

    for jobId, playerCount in pairs(serverCache) do

        if jobId
            and jobId ~= ""
            and jobId ~= currentJobId
            and not isBlockedJob(jobId)
        then

            playerCount =
                tonumber(playerCount) or 0

            -- =================================================
            -- ไม่เอา Server ที่เต็ม
            -- =================================================

            if playerCount < maxPlayers then

                table.insert(
                    validJobs,
                    {
                        jobId = jobId,
                        playerCount = playerCount
                    }
                )

            end
        end
    end

    return validJobs
end

-- =========================================================
-- SHUFFLE
-- =========================================================

local function shuffle(list)

    for i = #list, 2, -1 do

        local j =
            math.random(
                1,
                i
            )

        list[i], list[j] =
            list[j], list[i]
    end

    return list
end

-- =========================================================
-- SERVER HOP
-- =========================================================

local function serverHop()

    if getgenv().AntiAFKHopping then

        print(
            "[AntiAFK] Server Hop กำลังทำงานอยู่แล้ว"
        )

        return
    end

    getgenv().AntiAFKHopping = true
    getgenv().AntiAFKRunning = false

    print("================================")
    print("[AntiAFK] 🔄 SERVER CACHE HOP")
    print("[AntiAFK] Current JobId:", game.JobId)
    print("[AntiAFK] Blocked:", getBlockedJobCount())
    print("[AntiAFK] Cache:", getServerCacheCount())
    print("================================")

    -- =====================================================
    -- โหลด Cache ใหม่
    -- =====================================================

    loadServerCache()

    -- ลบ Blocked ออกจาก Cache
    removeBlockedJobsFromCache()

    -- =====================================================
    -- สร้างรายการ Server ที่ใช้ได้
    -- =====================================================

    local validJobs =
        getValidCachedJobs()

    if #validJobs == 0 then

        warn("================================")
        warn("[AntiAFK] ❌ ไม่มี Server ที่ใช้ได้")
        warn(
            "[AntiAFK] Cache:",
            getServerCacheCount()
        )
        warn(
            "[AntiAFK] Blocked:",
            getBlockedJobCount()
        )
        warn("================================")

        getgenv().AntiAFKHopping = false

        return false
    end

    -- =====================================================
    -- สุ่ม JobId
    -- =====================================================

    shuffle(validJobs)

    print(
        "[AntiAFK] 🎲 พบ Server ที่ใช้ได้:",
        #validJobs
    )

    -- =====================================================
    -- ลองตามลำดับที่สุ่ม
    -- =====================================================

    for _, serverInfo in ipairs(validJobs) do

        local serverJobId =
            serverInfo.jobId

        local playerCount =
            serverInfo.playerCount

        print("================================")
        print("[AntiAFK] 🎯 สุ่มเลือก Server")
        print("[AntiAFK] JobId:", serverJobId)
        print("[AntiAFK] Players:", playerCount)
        print("================================")

        local teleportFailed = false
        local teleportResult = nil
        local teleportMessage = nil

        local connection

        connection =
            TeleportService.TeleportInitFailed:
            Connect(
                function(
                    player,
                    result,
                    message
                )

                    if player ~= Player then
                        return
                    end

                    teleportFailed = true
                    teleportResult = result
                    teleportMessage = message
                end
            )

        -- =================================================
        -- TELEPORT
        -- =================================================

        local callSuccess,
            callError =
            pcall(function()

                TeleportService:
                    TeleportToPlaceInstance(
                        game.PlaceId,
                        serverJobId,
                        Player
                    )

            end)

        if not callSuccess then

            warn("================================")
            warn("[AntiAFK] ❌ Teleport Call Error")
            warn("[AntiAFK] JobId:", serverJobId)
            warn("[AntiAFK] Error:", callError)
            warn("================================")

            if connection then
                connection:Disconnect()
            end

            task.wait(
                HOP_RETRY_DELAY
            )

            continue
        end

        -- =================================================
        -- WAIT TELEPORT RESULT
        -- =================================================

        local timeout = 0

        while timeout < 10 do

            task.wait(0.25)

            timeout += 0.25

            if teleportFailed then
                break
            end
        end

        if connection then
            connection:Disconnect()
        end

        -- =================================================
        -- TELEPORT FAILED
        -- =================================================

        if teleportFailed then

            warn("================================")
            warn("[AntiAFK] ❌ Teleport ไม่สำเร็จ")
            warn("[AntiAFK] JobId:", serverJobId)
            warn(
                "[AntiAFK] Result:",
                tostring(teleportResult)
            )
            warn(
                "[AntiAFK] Message:",
                tostring(teleportMessage)
            )
            warn("[AntiAFK] 🔄 ลอง JobId ที่สุ่มตัวถัดไป")
            warn("================================")

            task.wait(
                HOP_RETRY_DELAY
            )

            continue
        end

        -- =================================================
        -- SUCCESS
        -- =================================================

        print("================================")
        print("[AntiAFK] 📡 ส่ง Teleport แล้ว")
        print("[AntiAFK] JobId:", serverJobId)
        print("[AntiAFK] Players:", playerCount)
        print("[AntiAFK] ⏳ รอเข้า Server ใหม่...")
        print("================================")

        return true
    end

    -- =====================================================
    -- ไม่มีตัวไหนเข้าได้
    -- =====================================================

    warn("================================")
    warn("[AntiAFK] ❌ ลองทุก JobId แล้วแต่เข้าไม่ได้")
    warn("================================")

    getgenv().AntiAFKHopping = false

    return false
end

-- =========================================================

local samePositionCount = 0
local lastPosition = nil

local function getPosition()

    local currentCharacter =
        Player.Character

    if not currentCharacter then
        return nil
    end

    local currentRoot =
        currentCharacter:
        FindFirstChild(
            "HumanoidRootPart"
        )

    if not currentRoot then
        return nil
    end

    return currentRoot.Position
end

-- =========================================================
-- POSITION CHECK
-- =========================================================

local function isSamePosition(
    pos1,
    pos2
)

    if not pos1 or not pos2 then
        return false
    end

    local distance =
        (pos1 - pos2).Magnitude

    print(
        "[AntiAFK] ระยะที่ขยับ:",
        math.floor(
            distance * 100
        ) / 100,
        "studs"
    )

    return distance <=
        POSITION_TOLERANCE
end
