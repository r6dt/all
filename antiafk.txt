if not game:IsLoaded() then game.Loaded:Wait() end

local Players = game:GetService("Players")
local TeleportService = game:GetService("TeleportService")
local HttpService = game:GetService("HttpService")

local Player = Players.LocalPlayer

local CHECK_INTERVAL = 60
local REQUIRED_CHECKS = 5
local POSITION_TOLERANCE = 5

local samePositionCount = 0
local lastPosition = nil

print("================================")
print("Anti-AFK Server Hop")
print("ทำงานแล้ว!")
print("Check ทุก:", CHECK_INTERVAL, "วินาที")
print("ต้องอยู่จุดเดิม:", REQUIRED_CHECKS, "ครั้ง")
print("Tolerance:", POSITION_TOLERANCE, "studs")
print("================================")

local function getPosition()
    local character = Player.Character

    if not character then
        return nil
    end

    local root = character:FindFirstChild("HumanoidRootPart")

    if not root then
        return nil
    end

    return root.Position
end

local function isSamePosition(pos1, pos2)
    if not pos1 or not pos2 then
        return false
    end

    local distance = (pos1 - pos2).Magnitude

    print("[AntiAFK] ระยะที่ขยับ:", distance, "studs")

    return distance <= POSITION_TOLERANCE
end

local function serverHop()
    print("[ServerHop] กำลังค้นหา Server ใหม่...")

    local placeId = game.PlaceId

    local success, response = pcall(function()
        return game:HttpGet(
            "https://games.roblox.com/v1/games/"
            .. placeId
            .. "/servers/Public?sortOrder=Asc&limit=100"
        )
    end)

    if not success then
        warn("[ServerHop] ไม่สามารถดึง Server List ได้:", response)
        return false
    end

    local decodeSuccess, data = pcall(function()
        return HttpService:JSONDecode(response)
    end)

    if not decodeSuccess or not data or not data.data then
        warn("[ServerHop] ไม่สามารถอ่าน Server List ได้")
        return false
    end

    for _, server in ipairs(data.data) do

        -- ไม่เข้าเซิร์ฟปัจจุบัน และต้องมีที่ว่าง
        if server.id ~= game.JobId
            and server.playing < server.maxPlayers then

            print("[ServerHop] พบ Server ใหม่:", server.id)
            print(
                "[ServerHop] Players:",
                server.playing .. "/" .. server.maxPlayers
            )

            local teleportSuccess, teleportError = pcall(function()
                TeleportService:TeleportToPlaceInstance(
                    placeId,
                    server.id,
                    Player
                )
            end)

            if teleportSuccess then
                print("[ServerHop] กำลัง Teleport...")
                return true
            else
                warn("[ServerHop] Teleport ไม่สำเร็จ:", teleportError)
                return false
            end
        end
    end

    warn("[ServerHop] ไม่พบ Server อื่น")
    return false
end

while true do
    task.wait(CHECK_INTERVAL)

    local currentPosition = getPosition()

    if currentPosition then

        if lastPosition and isSamePosition(lastPosition, currentPosition) then

            samePositionCount += 1

            print(
                "[AntiAFK] อยู่จุดเดิม | "
                .. samePositionCount
                .. "/"
                .. REQUIRED_CHECKS
            )

            if samePositionCount >= REQUIRED_CHECKS then

                print("================================")
                print(
                    "[AntiAFK] อยู่จุดเดิมครบ "
                    .. REQUIRED_CHECKS
                    .. " ครั้ง"
                )
                print("[ServerHop] กำลังหา Server ใหม่...")
                print("================================")

                if serverHop() then
                    break
                end
            end

        else

            samePositionCount = 1

            print(
                "[AntiAFK] ตำแหน่งเปลี่ยน -> Reset 1/"
                .. REQUIRED_CHECKS
            )

        end

        lastPosition = currentPosition
    end
end
