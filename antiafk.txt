-- =========================================================
-- LOAD JOB FILE
-- =========================================================

local function loadJobFile()

    if not hasFileAPI then
        return
    end

    if not isfile(JOB_FILE) then
        print("[AntiAFK] ยังไม่มีไฟล์", JOB_FILE)
        return
    end

    local success, content = pcall(function()
        return readfile(JOB_FILE)
    end)

    if not success then
        warn("[AntiAFK] อ่านไฟล์ JobId ไม่สำเร็จ:", content)
        return
    end

    if not content or content == "" then
        print("[AntiAFK] ไฟล์ JobId ว่าง")
        return
    end

    local count = 0

    for line in string.gmatch(content, "[^\r\n]+") do

        local jobId = line:match("^([^|]+)")

        if jobId and jobId ~= "" then

            if not blockedJobs[jobId] then
                blockedJobs[jobId] = true
                count += 1
            end

        end
    end

    print(
        "[AntiAFK] โหลด JobId จากไฟล์แล้ว:",
        count,
        "รายการ"
    )
end

loadJobFile()

-- =========================================================
-- SAVE JOB FILE
-- =========================================================

local function saveJobFile()

    if not hasFileAPI then
        warn("[AntiAFK] ไม่มี File API -> ไม่สามารถบันทึกไฟล์ได้")
        return false
    end

    local jobs = {}

    for jobId in pairs(blockedJobs) do
        table.insert(jobs, jobId)
    end

    while #jobs > MAX_BLOCKED_JOBS do
        table.remove(jobs, 1)
    end

    local output = ""

    for _, jobId in ipairs(jobs) do
        output = output .. jobId .. "\n"
    end

    local success, err = pcall(function()
        writefile(JOB_FILE, output)
    end)

    if not success then
        warn("[AntiAFK] บันทึก JobId ไม่สำเร็จ:", err)
        return false
    end

    print("[AntiAFK] บันทึก JobId ลง", JOB_FILE, "แล้ว")

    return true
end

-- =========================================================
-- ADD BLOCKED JOB
-- =========================================================

local function addBlockedJob(jobId)

    if not jobId or jobId == "" then
        warn("[AntiAFK] JobId ว่าง -> ไม่บันทึก")
        return false
    end

    if blockedJobs[jobId] then
        print("[AntiAFK] JobId นี้มีอยู่แล้ว:", jobId)
        return true
    end

    blockedJobs[jobId] = true

    print("================================")
    print("[AntiAFK] เพิ่ม JobId ที่ถูก Kick")
    print("[AntiAFK] JobId:", jobId)
    print("================================")

    local jobs = {}

    for id in pairs(blockedJobs) do
        table.insert(jobs, id)
    end

    while #jobs > MAX_BLOCKED_JOBS do

        local removeId = table.remove(jobs, 1)

        if removeId then
            blockedJobs[removeId] = nil
        end
    end

    return saveJobFile()
end

-- =========================================================
-- CHECK BLOCKED JOB
-- =========================================================

local function isBlockedJob(jobId)

    if not jobId or jobId == "" then
        return false
    end

    return blockedJobs[jobId] == true
end

-- =========================================================
-- COUNT BLOCKED JOBS
-- =========================================================

local function getBlockedJobCount()

    local count = 0

    for _ in pairs(blockedJobs) do
        count += 1
    end

    return count
end

-- =========================================================
-- PRINT JOB LIST
-- =========================================================

local function printBlockedJobs()

    local count = 0

    print("[AntiAFK] ===== Blocked JobIds =====")

    for jobId in pairs(blockedJobs) do

        count += 1

        print(
            "[AntiAFK]",
            count .. ".",
            jobId
        )
    end

    print(
        "[AntiAFK] ทั้งหมด:",
        count,
        "JobId"
    )

    print("[AntiAFK] =========================")
end

printBlockedJobs()

-- =========================================================
-- SERVER HOP
-- RANDOM + BLOCKED JOB FILTER + 771 RETRY
-- =========================================================

local function serverHop()

    if getgenv().AntiAFKHopping then

        print(
            "[AntiAFK] Server Hop กำลังทำงานอยู่แล้ว"
        )

        return
    end

    getgenv().AntiAFKHopping = true
    getgenv().AntiAFKRunning = false

    local placeId = game.PlaceId

    print("================================")
    print("[AntiAFK] 🔄 RANDOM SERVER HOP")
    print("[AntiAFK] Current JobId:", game.JobId)
    print("[AntiAFK] Blocked Jobs:", getBlockedJobCount())
    print("[AntiAFK] Retry Delay:", HOP_RETRY_DELAY, "วินาที")
    print("[AntiAFK] Page Limit:", SERVER_PAGE_LIMIT)
    print("================================")

    -- =====================================================
    -- TELEPORT FAILED EVENT
    -- =====================================================

    local teleportFailed = false
    local teleportFailResult = nil
    local teleportFailMessage = nil
    local teleportFailJobId = nil

    local teleportConnection

    teleportConnection =
        TeleportService.TeleportInitFailed:Connect(
            function(
                player,
                teleportResult,
                errorMessage
            )

                if player ~= Player then
                    return
                end

                teleportFailed = true
                teleportFailResult = teleportResult
                teleportFailMessage = errorMessage

                warn("================================")
                warn("[AntiAFK] ❌ TELEPORT FAILED")
                warn(
                    "[AntiAFK] Result:",
                    tostring(teleportResult)
                )
                warn(
                    "[AntiAFK] Message:",
                    tostring(errorMessage)
                )

                if teleportFailJobId then

                    warn(
                        "[AntiAFK] ❌ JobId ที่เข้าไม่ได้:",
                        teleportFailJobId
                    )

                end

                -- =================================================
                -- 771 CHECK
                -- =================================================

                local resultText =
                    tostring(teleportResult)

                local messageText =
                    tostring(errorMessage)

                if string.find(resultText, "771", 1, true)
                    or string.find(messageText, "771", 1, true)
                then

                    warn(
                        "[AntiAFK] ⚠️ ตรวจพบ Error 771"
                    )

                    warn(
                        "[AntiAFK] 🔄 จะข้าม JobId นี้และสุ่ม Server ใหม่"
                    )

                end

                warn("================================")
            end
        )

    -- =====================================================
    -- SEARCH LOOP
    -- =====================================================

    while true do

        local currentJobId = game.JobId

        -- =================================================
        -- SERVERS TRIED IN THIS SEARCH ROUND
        -- =================================================

        local triedServers = {}

        triedServers[currentJobId] = true

        -- =================================================
        -- COLLECT SERVER LIST
        -- =================================================

        local serverCandidates = {}

        local cursor = nil

        for page = 1, SERVER_PAGE_LIMIT do

            print(
                "[AntiAFK] 🔎 กำลังค้นหา Server... Page:",
                page
            )

            local url =
                "https://games.roblox.com/v1/games/"
                .. tostring(placeId)
                .. "/servers/Public?sortOrder=Asc&limit=100"

            if cursor then

                url =
                    url
                    .. "&cursor="
                    .. HttpService:UrlEncode(cursor)

            end

            -- =================================================
            -- GET SERVER LIST
            -- =================================================

            local success, result =
                pcall(function()

                    return game:HttpGet(url)

                end)

            if not success then

                warn(
                    "[AntiAFK] ❌ ดึง Server List ไม่สำเร็จ:",
                    result
                )

                break
            end

            -- =================================================
            -- JSON
            -- =================================================

            local decodeSuccess, data =
                pcall(function()

                    return HttpService:JSONDecode(result)

                end)

            if not decodeSuccess then

                warn(
                    "[AntiAFK] ❌ Decode Server List ไม่สำเร็จ:",
                    data
                )

                break
            end

            if not data or not data.data then

                warn(
                    "[AntiAFK] ❌ Server List ไม่มีข้อมูล"
                )

                break
            end

            print(
                "[AntiAFK] พบ Server Page นี้:",
                #data.data
            )

            -- =================================================
            -- COLLECT AVAILABLE SERVERS
            -- =================================================

            for _, server in ipairs(data.data) do

                local serverJobId =
                    server.id

                local playing =
                    tonumber(server.playing) or 0

                local maxPlayers =
                    tonumber(server.maxPlayers) or 0

                if serverJobId
                    and serverJobId ~= ""
                    and serverJobId ~= currentJobId
                    and not triedServers[serverJobId]
                    and not isBlockedJob(serverJobId)
                then

                    local freeSlots =
                        maxPlayers - playing

                    if maxPlayers > SERVER_SLOT_BUFFER
                        and freeSlots > SERVER_SLOT_BUFFER
                    then

                        table.insert(
                            serverCandidates,
                            {
                                id = serverJobId,
                                playing = playing,
                                maxPlayers = maxPlayers,
                                freeSlots = freeSlots
                            }
                        )

                    end
                end
            end

            -- =================================================
            -- NEXT PAGE
            -- =================================================

            cursor =
                data.nextPageCursor

            if not cursor then

                print(
                    "[AntiAFK] ไม่มีหน้า Server เพิ่มแล้ว"
                )

                break
            end
        end

        -- =====================================================
        -- SERVER CANDIDATE COUNT
        -- =====================================================

        print(
            "[AntiAFK] 🎲 Server ที่ผ่าน Filter:",
            #serverCandidates
        )

        -- =====================================================
        -- RANDOMIZE SERVER LIST
        -- =====================================================

        for i = #serverCandidates, 2, -1 do

            local randomIndex =
                math.random(1, i)

            serverCandidates[i],
            serverCandidates[randomIndex] =
                serverCandidates[randomIndex],
                serverCandidates[i]
        end

        -- =====================================================
        -- RANDOM SERVER TELEPORT
        -- =====================================================

        for _, server in ipairs(serverCandidates) do

            local serverJobId =
                server.id

            if not triedServers[serverJobId] then

                triedServers[serverJobId] = true

                -- =================================================
                -- RESET TELEPORT STATUS
                -- =================================================

                teleportFailed = false
                teleportFailResult = nil
                teleportFailMessage = nil
                teleportFailJobId = serverJobId

                print("================================")
                print("[AntiAFK] 🎲 RANDOM SERVER")
                print("[AntiAFK] JobId:", serverJobId)

                print(
                    "[AntiAFK] Players:",
                    server.playing
                    .. "/"
                    .. server.maxPlayers
                )

                print(
                    "[AntiAFK] เหลือช่อง:",
                    server.freeSlots
                )

                print(
                    "[AntiAFK] 🚀 กำลัง Teleport..."
                )

                print("================================")

                -- =================================================
                -- TELEPORT
                -- =================================================

                local teleportCallSuccess,
                    teleportCallError =
                    pcall(function()

                        TeleportService:
                            TeleportToPlaceInstance(
                                placeId,
                                serverJobId,
                                Player
                            )

                    end)

                if not teleportCallSuccess then

                    warn("================================")
                    warn(
                        "[AntiAFK] ❌ Teleport Call Error"
                    )

                    warn(
                        "[AntiAFK] Error:",
                        tostring(teleportCallError)
                    )

                    warn(
                        "[AntiAFK] ❌ ข้าม JobId:",
                        serverJobId
                    )

                    warn(
                        "[AntiAFK] 🔄 สุ่ม Server ใหม่..."
                    )

                    warn("================================")

                    teleportFailed = false
                    teleportFailResult = nil
                    teleportFailMessage = nil
                    teleportFailJobId = nil

                    task.wait(0.5)

                    continue
                end

                print(
                    "[AntiAFK] 📡 ส่ง Teleport แล้ว"
                )

                -- =================================================
                -- WAIT TELEPORT RESULT
                -- =================================================

                local waitTime = 0

                local TELEPORT_TIMEOUT = 10

                while waitTime < TELEPORT_TIMEOUT do

                    task.wait(0.25)

                    waitTime += 0.25

                    if teleportFailed then
                        break
                    end
                end

                -- =================================================
                -- TELEPORT FAILED
                -- =================================================

                if teleportFailed then

                    warn("================================")
                    warn(
                        "[AntiAFK] ❌ Teleport ล้มเหลว"
                    )

                    warn(
                        "[AntiAFK] Result:",
                        tostring(
                            teleportFailResult
                        )
                    )

                    warn(
                        "[AntiAFK] Message:",
                        tostring(
                            teleportFailMessage
                        )
                    )

                    warn(
                        "[AntiAFK] ❌ ข้าม JobId:",
                        serverJobId
                    )

                    warn(
                        "[AntiAFK] 🎲 สุ่ม Server ถัดไป..."
                    )

                    warn("================================")

                    teleportFailed = false
                    teleportFailResult = nil
                    teleportFailMessage = nil
                    teleportFailJobId = nil

                    task.wait(0.5)

                    continue
                end

                -- =================================================
                -- NO TELEPORT ERROR
                -- =================================================

                print(
                    "[AntiAFK] ⏳ ยังไม่มี Teleport Error"
                )

                print(
                    "[AntiAFK] 📡 รอการย้าย Server..."
                )

                -- =================================================
                -- สำคัญ
                --
                -- ถ้า Teleport สำเร็จจริง
                -- Script จะเริ่มใหม่ใน Server ใหม่
                --
                -- ถ้าไม่มี Error ภายใน Timeout
                -- ไม่ควรยิง TP ซ้ำทันที
                -- =================================================

                task.wait(1)

                return
            end
        end

        -- =====================================================
        -- NO SERVER FOUND
        -- =====================================================

        warn(
            "[AntiAFK] ❌ ไม่มี Server ที่สามารถเข้าได้"
        )

        print(
            "[AntiAFK] 🔄 จะค้นหาใหม่ใน",
            HOP_RETRY_DELAY,
            "วินาที..."
        )

        task.wait(HOP_RETRY_DELAY)
    end
end

-- =========================================================
-- POSITION
-- =========================================================

local samePositionCount = 0
local lastPosition = nil

local function getPosition()

    local currentCharacter =
        Player.Character

    if not currentCharacter then
        return nil
    end

    local currentRoot =
        currentCharacter:FindFirstChild(
            "HumanoidRootPart"
        )

    if not currentRoot then
        return nil
    end

    return currentRoot.Position
end

-- =========================================================
-- POSITION CHECK
-- =========================================================

local function isSamePosition(
    pos1,
    pos2
)

    if not pos1 or not pos2 then
        return false
    end

    local distance =
        (pos1 - pos2).Magnitude

    print(
        "[AntiAFK] ระยะที่ขยับ:",
        math.floor(distance * 100) / 100,
        "studs"
    )

    return distance <= POSITION_TOLERANCE
end

print("[AntiAFK] Main Loop เริ่มทำงานแล้ว")

    task.wait(CHECK_INTERVAL)

    if not getgenv().AntiAFKRunning then

    end

        warn(
            "[AntiAFK] ไม่พบ Character/HumanoidRootPart"

        )

    end

    -- =====================================================

        lastPosition =
            currentPosition

        samePositionCount = 0

        print(
            "[AntiAFK] บันทึกตำแหน่งเริ่มต้นแล้ว"
        )

            "[AntiAFK] Check: 0/"
            .. REQUIRED_CHECKS
        )

    -- =====================================================
    -- SAME POSITION
    -- =====================================================

    elseif isSamePosition(
        lastPosition,
        currentPosition
    ) then

        samePositionCount += 1

        print(
            "[AntiAFK] อยู่จุดเดิม | "
            .. samePositionCount
            .. "/"
            .. REQUIRED_CHECKS
        )

        -- =================================================
        -- AFK TRIGGER
        -- =================================================

        if samePositionCount >= REQUIRED_CHECKS then

            print("================================")
            print("[AntiAFK] ⚠️ AFK TRIGGERED")

            print(
                "[AntiAFK] นิ่งครบ "
                .. REQUIRED_CHECKS
                .. " ครั้ง"
            )

            local currentJobId =
                game.JobId

            print(
                "[AntiAFK] กำลังบันทึก JobId..."
            )

            local saved =
                addBlockedJob(
                    currentJobId
                )

            if saved then

                print(
                    "[AntiAFK] ✅ JobId ถูกบันทึกแล้ว"
                )

                print(
                    "[AntiAFK] JobId:",
                    currentJobId
                )

            else

                warn(
                    "[AntiAFK] ❌ บันทึก JobId ไม่สำเร็จ"
                )
            end

            print(
                "[AntiAFK] 🔴 สาเหตุ: AFK"
            )

            print(
                "[AntiAFK] กำลังเตะออกจาก Server..."
            )

            print("================================")

            getgenv().AntiAFKRunning =
                false

            task.wait(1)

            Player:Kick(
                "AntiAFK: AFK Server"
            )

            break
        end

    -- =====================================================
    -- POSITION CHANGED
    -- =====================================================

    else

        samePositionCount = 0

        print(
            "[AntiAFK] ตำแหน่งเปลี่ยน"
            .. " -> Reset 0/"
            .. REQUIRED_CHECKS
        )
    end

    lastPosition =
        currentPosition
end
