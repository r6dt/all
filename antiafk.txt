-- =========================================================
-- LOAD BLOCKED JOB FILE
-- =========================================================

local function loadJobFile()

    if not hasFileAPI then
        return
    end

    if not isfile(JOB_FILE) then

        print(
            "[AntiAFK] ยังไม่มี",
            JOB_FILE
        )

        return
    end

    local success, content =
        pcall(function()
            return readfile(JOB_FILE)
        end)

    if not success then

        warn(
            "[AntiAFK] อ่าน Job File ไม่สำเร็จ:",
            content
        )

        return
    end

    if not content or content == "" then
        return
    end

    local count = 0

    for line in string.gmatch(
        content,
        "[^\r\n]+"
    ) do

        local jobId =
            line:match("^([^|]+)")

        if jobId then

            jobId =
                jobId:match(
                    "^%s*(.-)%s*$"
                )

        end

        if jobId
            and jobId ~= ""
        then

            if not blockedJobs[jobId] then

                blockedJobs[jobId] = true
                count += 1

            end
        end
    end

    print(
        "[AntiAFK] โหลด Blocked JobId:",
        count
    )
end

loadJobFile()

-- =========================================================
-- CHECK BLOCKED JOB
-- =========================================================

local function isBlockedJob(jobId)

    if not jobId
        or jobId == ""
    then
        return false
    end

    return blockedJobs[jobId] == true
end

-- =========================================================
-- COUNT BLOCKED
-- =========================================================

local function getBlockedJobCount()

    local count = 0

    for _ in pairs(blockedJobs) do
        count += 1
    end

    return count
end

-- =========================================================
-- SAVE BLOCKED JOB FILE
-- =========================================================

local function saveJobFile()

    if not hasFileAPI then
        return false
    end

    local jobs = {}

    for jobId in pairs(blockedJobs) do

        table.insert(
            jobs,
            jobId
        )

    end

    -- =====================================================
    -- จำกัดจำนวน Blocked สูงสุด
    -- =====================================================

    while #jobs > MAX_BLOCKED_JOBS do

        local removeId =
            table.remove(
                jobs,
                1
            )

        if removeId then
            blockedJobs[removeId] = nil
        end
    end

    local output = ""

    for _, jobId in ipairs(jobs) do

        output =
            output
            .. jobId
            .. "\n"

    end

    local success, err =
        pcall(function()

            writefile(
                JOB_FILE,
                output
            )

        end)

    if not success then

        warn(
            "[AntiAFK] บันทึก Blocked Job ไม่สำเร็จ:",
            err
        )

        return false
    end

    return true
end

end

-- =========================================================
-- SAVE SERVER CACHE
-- =========================================================

local function saveServerCache()

    if not hasFileAPI then
        return false
    end

    local now =
        os.time()

    local output = ""
    local count = 0

    local written = {}

    for jobId, data in pairs(serverCache) do

        if not written[jobId]
            and data
            and not isBlockedJob(jobId)
        then

            local timestamp =
                tonumber(
                    data.timestamp
                ) or 0

            -- =================================================
            -- เขียนเฉพาะ Cache ที่ยังไม่หมดอายุ
            -- =================================================

            if timestamp > 0
                and now - timestamp
                    < SERVER_CACHE_EXPIRE_TIME
            then

                written[jobId] = true

                local players =
                    tonumber(
                        data.players
                    ) or 0

                local maxPlayers =
                    tonumber(
                        data.maxPlayers
                    ) or 0

                output =
                    output
                    .. jobId
                    .. "|"
                    .. tostring(players)
                    .. "|"
                    .. tostring(maxPlayers)
                    .. "|"
                    .. tostring(timestamp)
                    .. "\n"

                count += 1
            end
        end
    end

    local success, err =
        pcall(function()

            writefile(
                SERVER_CACHE_FILE,
                output
            )

        end)

    if not success then

        warn(
            "[AntiAFK] ❌ บันทึก Server Cache ไม่สำเร็จ:",
            err
        )

        return false
    end

    print(
        "[AntiAFK] 💾 Server Cache:",
        count,
        "JobId"
    )

    return true
end

-- =========================================================
-- LOAD SERVER CACHE
-- =========================================================

local function loadServerCache()

    serverCache = {}

    if not hasFileAPI then

        warn(
            "[AntiAFK] File API ไม่มี -> โหลด Cache ไม่ได้"
        )

        return false
    end

    if not isfile(
        SERVER_CACHE_FILE
    ) then

        print(
            "[AntiAFK] ยังไม่มี",
            SERVER_CACHE_FILE
        )

        return false
    end

    local success, content =
        pcall(function()

            return readfile(
                SERVER_CACHE_FILE
            )

        end)

    if not success then

        warn(
            "[AntiAFK] อ่าน Server Cache ไม่สำเร็จ:",
            content
        )

        return false
    end

    if not content
        or content == ""
    then

        print(
            "[AntiAFK] server_cache.txt ว่าง"
        )

        return false
    end

    local now =
        os.time()

    local count = 0
    local duplicateCount = 0
    local blockedCount = 0
    local expiredCount = 0

    local fileNeedsCleanup = false

    -- =====================================================
    -- อ่านแต่ละบรรทัด
    -- =====================================================

    for line in string.gmatch(
        content,
        "[^\r\n]+"
    ) do

        local jobId,
            playerCount,
            maxPlayers,
            timestamp =
            line:match(
                "^([^|]+)|([^|]+)|([^|]+)|([^|]+)"
            )

        if jobId then

            jobId =
                jobId:match(
                    "^%s*(.-)%s*$"
                )

        end

        playerCount =
            tonumber(
                playerCount
            ) or 0

        maxPlayers =
            tonumber(
                maxPlayers
            ) or 0

        timestamp =
            tonumber(
                timestamp
            ) or 0

        -- =================================================
        -- INVALID FORMAT
        -- =================================================

        if not jobId
            or jobId == ""
            or timestamp <= 0
        then

            fileNeedsCleanup = true

        else

            -- =================================================
            -- BLOCKED
            -- =================================================

            if isBlockedJob(jobId) then

                blockedCount += 1
                fileNeedsCleanup = true

            -- =================================================
            -- EXPIRED
            -- =================================================

            elseif now - timestamp
                >= SERVER_CACHE_EXPIRE_TIME
            then

                expiredCount += 1
                fileNeedsCleanup = true

                print(
                    "[AntiAFK] ⏰ Cache หมดอายุ:",
                    jobId,
                    "| อายุ:",
                    math.floor(
                        (now - timestamp) / 60
                    ),
                    "นาที"
                )

            -- =================================================
            -- DUPLICATE
            -- =================================================

            elseif serverCache[jobId] then

                duplicateCount += 1
                fileNeedsCleanup = true

                if timestamp >
                    serverCache[jobId].timestamp
                then

                    serverCache[jobId] = {

                        players =
                            playerCount,

                        maxPlayers =
                            maxPlayers,

                        timestamp =
                            timestamp
                    }
                end

            -- =================================================
            -- NEW
            -- =================================================

            else

                serverCache[jobId] = {

                    players =
                        playerCount,

                    maxPlayers =
                        maxPlayers,

                    timestamp =
                        timestamp
                }

                count += 1
            end
        end
    end

    print("================================")
    print("[AntiAFK] 📦 โหลด Server Cache")
    print("[AntiAFK] Valid:", count)
    print("[AntiAFK] Duplicate:", duplicateCount)
    print("[AntiAFK] Blocked:", blockedCount)
    print("[AntiAFK] Expired:", expiredCount)
    print("================================")

    if fileNeedsCleanup then

        print(
            "[AntiAFK] 🧹 กำลัง Clean server_cache.txt..."
        )

        saveServerCache()

    end

    return count > 0
end

-- =========================================================
-- REMOVE EXPIRED / BLOCKED CACHE
-- =========================================================

local function removeExpiredCache()

    local now =
        os.time()

    local removedExpired = 0
    local removedBlocked = 0

    for jobId, data in pairs(serverCache) do

        if isBlockedJob(jobId) then

            serverCache[jobId] = nil
            removedBlocked += 1

        else

            local timestamp =
                tonumber(
                    data.timestamp
                ) or 0

            if timestamp <= 0
                or now - timestamp
                    >= SERVER_CACHE_EXPIRE_TIME
            then

                serverCache[jobId] = nil
                removedExpired += 1

            end
        end
    end

    if removedExpired > 0 then

        print(
            "[AntiAFK] ⏰ ลบ Cache หมดอายุ:",
            removedExpired
        )

    end

    if removedBlocked > 0 then

        print(
            "[AntiAFK] 🚫 ลบ Blocked จาก Cache:",
            removedBlocked
        )

    end

    if removedExpired > 0
        or removedBlocked > 0
    then

        saveServerCache()

    end
end

-- =========================================================
-- GET VALID CACHED JOBS
-- =========================================================

local function getValidCachedJobs()

    local now =
        os.time()

    local currentJobId =
        game.JobId

    local candidates = {}

    for jobId, data in pairs(
        serverCache
    ) do

        if jobId
            and jobId ~= ""
            and jobId ~= currentJobId
            and not isBlockedJob(jobId)
            and data
        then

            local players =
                tonumber(
                    data.players
                ) or 0

            local maxPlayers =
                tonumber(
                    data.maxPlayers
                ) or 0

            local timestamp =
                tonumber(
                    data.timestamp
                ) or 0

            if timestamp > 0 then

                local age =
                    now - timestamp

                if age >= 0
                    and age <
                        SERVER_CACHE_EXPIRE_TIME
                then

                    if maxPlayers <= 0
                        or players < maxPlayers
                    then

                        table.insert(
                            candidates,
                            {

                                jobId =
                                    jobId,

                                players =
                                    players,

                                maxPlayers =
                                    maxPlayers,

                                timestamp =
                                    timestamp,

                                age =
                                    age
                            }
                        )

                    end
                end
            end
        end
    end

    return candidates
end

-- =========================================================
-- GET LIVE SERVER LIST
-- =========================================================

local function getLiveServers()

    local servers = {}

    if not hasHttpAPI then

        warn(
            "[AntiAFK] ❌ HTTP API ไม่พร้อมใช้งาน"
        )

        return servers
    end

    local currentJobId =
        game.JobId

    local cursor = ""
    local page = 0

    print("================================")
    print("[AntiAFK] 🌐 กำลังดึง Live Server List")
    print("[AntiAFK] PlaceId:", game.PlaceId)
    print("================================")

    while page < LIVE_SERVER_MAX_PAGES do

        page += 1

        local url =
            "https://games.roblox.com/v1/games/"
            .. tostring(game.PlaceId)
            .. "/servers/Public?sortOrder=Asc&limit="
            .. tostring(LIVE_SERVER_LIMIT)

        if cursor
            and cursor ~= ""
        then

            url =
                url
                .. "&cursor="
                .. HttpService:UrlEncode(
                    cursor
                )
        end

        local success, response =
            pcall(function()

                return game:HttpGet(
                    url
                )

            end)

        if not success then

            warn(
                "[AntiAFK] ❌ Live Server Request Error:",
                response
            )

            break
        end

        local decodeSuccess, data =
            pcall(function()

                return HttpService:JSONDecode(
                    response
                )

            end)

        if not decodeSuccess then

            warn(
                "[AntiAFK] ❌ JSON Decode Error:",
                data
            )

            break
        end

        if type(data) ~= "table" then

            warn(
                "[AntiAFK] ❌ Live Server Response ไม่ถูกต้อง"
            )

            break
        end

        if type(data.data) == "table" then

            for _, server in ipairs(
                data.data
            ) do

                if type(server) == "table" then

                    local serverId =
                        server.id

                    local playing =
                        tonumber(
                            server.playing
                        ) or 0

                    local maxPlayers =
                        tonumber(
                            server.maxPlayers
                        ) or 0

                    if serverId
                        and serverId ~= ""
                        and serverId ~= currentJobId
                        and not isBlockedJob(serverId)
                    then

                        if maxPlayers <= 0
                            or playing < maxPlayers
                        then

                            table.insert(
                                servers,
                                {

                                    jobId =
                                        serverId,

                                    players =
                                        playing,

                                    maxPlayers =
                                        maxPlayers,

                                    fps =
                                        tonumber(
                                            server.fps
                                        ) or 0,

                                    ping =
                                        tonumber(
                                            server.ping
                                        ) or 0
                                }
                            )

                        end
                    end
                end
            end
        end

        print(
            "[AntiAFK] Live Page:",
            page,
            "| Servers:",
            #servers
        )

        cursor =
            data.nextPageCursor

        if not cursor
            or cursor == ""
        then

            break
        end
    end

    -- =====================================================
    -- SORT LIVE SERVER
    -- คนน้อย -> คนเยอะ
    -- =====================================================

    table.sort(
        servers,
        function(a, b)

            if a.players ~= b.players then

                return a.players < b.players

            end

            if a.ping ~= b.ping
                and a.ping > 0
                and b.ping > 0
            then

                return a.ping < b.ping
            end

            return a.jobId < b.jobId
        end
    )

    print("================================")
    print(
        "[AntiAFK] ✅ Live Server ทั้งหมด:",
        #servers
    )
    print(
        "[AntiAFK] 📊 เรียงจากคนน้อย -> คนเยอะ"
    )
    print("================================")

    return servers
end

-- =========================================================
-- SERVER HOP
-- =========================================================

local function serverHop()

    if getgenv().AntiAFKHopping then

        print(
            "[AntiAFK] Server Hop กำลังทำงานอยู่แล้ว"
        )

        return false
    end

    getgenv().AntiAFKHopping = true
    getgenv().AntiAFKRunning = false

    print("================================")
    print("[AntiAFK] 🔄 LIVE SERVER HOP")
    print("[AntiAFK] Mode: LOW PLAYER FIRST")
    print(
        "[AntiAFK] Current JobId:",
        game.JobId
    )
    print(
        "[AntiAFK] Blocked:",
        getBlockedJobCount()
    )
    print("================================")

    -- =====================================================
    -- โหลด Cache ไว้เหมือนเดิม
    -- =====================================================

    loadServerCache()

    removeExpiredCache()

    -- =====================================================
    -- LIVE SERVER LIST
    -- =====================================================

    local liveServers =
        getLiveServers()

    if #liveServers == 0 then

        warn("================================")
        warn(
            "[AntiAFK] ❌ ไม่พบ Live Server ที่ใช้ได้"
        )
        warn(
            "[AntiAFK] Blocked:",
            getBlockedJobCount()
        )
        warn("================================")

        getgenv().AntiAFKHopping =
            false

        return false
    end

    -- =====================================================
    -- แสดง Server ที่คนน้อยที่สุด
    -- =====================================================

    print("================================")
    print("[AntiAFK] 📊 LIVE SERVER ORDER")
    print("================================")

    local displayCount =
        math.min(
            #liveServers,
            20
        )

    for index = 1, displayCount do

        local serverInfo =
            liveServers[index]

        print(
            "[AntiAFK] #"
            .. index
            .. " | Players: "
            .. serverInfo.players
            .. "/"
            .. serverInfo.maxPlayers
            .. " | Ping: "
            .. serverInfo.ping
            .. " | JobId: "
            .. serverInfo.jobId
        )
    end

    print("================================")

    -- =====================================================
    -- ลองทีละ Server
    -- คนน้อย -> คนเยอะ
    -- =====================================================

    for index, serverInfo in ipairs(
        liveServers
    ) do

        local serverJobId =
            serverInfo.jobId

        -- =================================================
        -- ตรวจอีกครั้งก่อน Teleport
        -- =================================================

        if serverJobId == game.JobId then

            continue
        end

        if isBlockedJob(
            serverJobId
        ) then

            print(
                "[AntiAFK] 🚫 ข้าม Blocked JobId:",
                serverJobId
            )

            continue
        end

        print("================================")

        print(
            "[AntiAFK] 🎯 Low Player Server:",
            index
            .. "/"
            .. #liveServers
        )

        print(
            "[AntiAFK] JobId:",
            serverJobId
        )

        print(
            "[AntiAFK] Players:",
            serverInfo.players
            .. "/"
            .. serverInfo.maxPlayers
        )

        print(
            "[AntiAFK] Ping:",
            serverInfo.ping
        )

        print("================================")

        local teleportFailed =
            false

        local teleportResult =
            nil

        local teleportMessage =
            nil

        local connection

        connection =
            TeleportService
            .TeleportInitFailed
            :Connect(
                function(
                    player,
                    result,
                    message
                )

                    if player ~= Player then
                        return
                    end

                    teleportFailed =
                        true

                    teleportResult =
                        result

                    teleportMessage =
                        message
                end
            )

        -- =================================================
        -- TELEPORT
        -- =================================================

        local callSuccess,
            callError =
            pcall(function()

                TeleportService:
                    TeleportToPlaceInstance(
                        game.PlaceId,
                        serverJobId,
                        Player
                    )

            end)

        if not callSuccess then

            warn(
                "[AntiAFK] ❌ Teleport Error:",
                callError
            )

            if connection then
                connection:Disconnect()
            end

            task.wait(
                HOP_RETRY_DELAY
            )

            continue
        end

        -- =================================================
        -- WAIT TELEPORT RESULT
        -- =================================================

        local timeout =
            0

        while timeout <
            TELEPORT_TIMEOUT
        do

            task.wait(0.25)

            timeout += 0.25

            if teleportFailed then
                break
            end
        end

        if connection then
            connection:Disconnect()
        end

        -- =================================================
        -- TELEPORT FAILED
        -- =================================================

        if teleportFailed then

            warn("================================")

            warn(
                "[AntiAFK] ❌ เข้า Server ไม่สำเร็จ"
            )

            warn(
                "[AntiAFK] JobId:",
                serverJobId
            )

-- =========================================================
-- GET POSITION
-- =========================================================

local function getPosition()

    local currentCharacter =
        Player.Character

    if not currentCharacter then
        return nil
    end

    local currentRoot =
        currentCharacter:
        FindFirstChild(
            "HumanoidRootPart"
        )

    if not currentRoot then
        return nil
    end

    return currentRoot.Position
end

-- =========================================================
-- CHECK SAME POSITION
-- =========================================================

local function isSamePosition(
    pos1,
    pos2
)

    if not pos1
        or not pos2
    then

        return false
    end

    local distance =
        (pos1 - pos2).Magnitude

    print(
        "[AntiAFK] ระยะที่ขยับ:",
        math.floor(
            distance * 100
        ) / 100,
        "studs"
    )

    return distance <=
        POSITION_TOLERANCE
end
