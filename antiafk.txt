    getgenv().AntiAFKBlockedJobs = {}
end

local blockedJobs = getgenv().AntiAFKBlockedJobs

-- =========================================================
-- LOAD JOB FILE
-- =========================================================

local function loadJobFile()

    if not hasFileAPI then
        return
    end

    if not isfile(JOB_FILE) then
        print("[AntiAFK] ยังไม่มีไฟล์", JOB_FILE)
        return
    end

    local success, content = pcall(function()
        return readfile(JOB_FILE)
    end)

    if not success then
        warn("[AntiAFK] อ่านไฟล์ JobId ไม่สำเร็จ:", content)
        return
    end

    if not content or content == "" then
        print("[AntiAFK] ไฟล์ JobId ว่าง")
        return
    end

    local count = 0

    for line in string.gmatch(content, "[^\r\n]+") do

        local jobId = line:match("^([^|]+)")

        if jobId and jobId ~= "" then
            blockedJobs[jobId] = true
            count += 1
        end
    end

    print("[AntiAFK] โหลด JobId จากไฟล์แล้ว:", count, "รายการ")
end

loadJobFile()

-- =========================================================
-- SAVE JOB FILE
-- =========================================================

local function saveJobFile()

    if not hasFileAPI then
        warn("[AntiAFK] ไม่มี File API -> ไม่สามารถบันทึกไฟล์ได้")
        return false
    end

    local jobs = {}

    for jobId in pairs(blockedJobs) do
        table.insert(jobs, jobId)
    end

    -- จำกัดจำนวน JobId
    while #jobs > MAX_BLOCKED_JOBS do
        table.remove(jobs, 1)
    end

    local output = ""

    for _, jobId in ipairs(jobs) do
        output = output .. jobId .. "\n"
    end

    local success, err = pcall(function()
        writefile(JOB_FILE, output)
    end)

    if not success then
        warn("[AntiAFK] บันทึก JobId ไม่สำเร็จ:", err)
        return false
    end

    print(
        "[AntiAFK] บันทึก JobId ลง",
        JOB_FILE,
        "แล้ว"
    )

    return true
end

-- =========================================================
-- ADD BLOCKED JOB
-- =========================================================

local function addBlockedJob(jobId)

    if not jobId or jobId == "" then
        warn("[AntiAFK] JobId ว่าง -> ไม่บันทึก")
        return false
    end

    if blockedJobs[jobId] then
        print("[AntiAFK] JobId นี้มีอยู่แล้ว:", jobId)
        return true
    end

    blockedJobs[jobId] = true

    print("================================")
    print("[AntiAFK] เพิ่ม JobId ที่ถูก Kick")
    print("[AntiAFK] JobId:", jobId)
    print("================================")

    -- จำกัดจำนวน JobId
    local jobs = {}

    for id in pairs(blockedJobs) do
        table.insert(jobs, id)
    end

    while #jobs > MAX_BLOCKED_JOBS do

        local removeId = table.remove(jobs, 1)

        if removeId then
            blockedJobs[removeId] = nil
        end
    end

    return saveJobFile()
end

-- =========================================================
-- CHECK BLOCKED JOB
-- =========================================================

local function isBlockedJob(jobId)

    if not jobId or jobId == "" then
        return false
    end

    return blockedJobs[jobId] == true
end

-- =========================================================
-- PRINT JOB LIST
-- =========================================================

local function printBlockedJobs()

    local count = 0

    print("[AntiAFK] ===== Blocked JobIds =====")

    for jobId in pairs(blockedJobs) do

        count += 1

        print(
            "[AntiAFK]",
            count .. ".",
            jobId
        )
    end

    print(
        "[AntiAFK] ทั้งหมด:",
        count,
        "JobId"
    )

    print("[AntiAFK] =========================")
end

-- =========================================================
-- POSITION
-- =========================================================

local samePositionCount = 0
local lastPosition = nil

local function getPosition()

    local character = Player.Character

    if not character then
        return nil
    end

    local root =
        character:FindFirstChild("HumanoidRootPart")

    if not root then
        return nil
    end

    return root.Position
end

-- =========================================================
-- POSITION CHECK
-- =========================================================

local function isSamePosition(pos1, pos2)

    if not pos1 or not pos2 then
        return false
    end

    local distance =
        (pos1 - pos2).Magnitude

    print(
        "[AntiAFK] ระยะที่ขยับ:",
        math.floor(distance * 100) / 100,
        "studs"
    )

    return distance <= POSITION_TOLERANCE
end
