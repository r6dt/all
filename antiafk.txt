-- =========================================================
-- LOAD JOB FILE
-- =========================================================

local function loadJobFile()

    if not hasFileAPI then
        return
    end

    if not isfile(JOB_FILE) then
        print("[AntiAFK] ยังไม่มีไฟล์", JOB_FILE)
        return
    end

    local success, content = pcall(function()
        return readfile(JOB_FILE)
    end)

    if not success then
        warn("[AntiAFK] อ่าน Job File ไม่สำเร็จ:", content)
        return
    end

    if not content or content == "" then
        return
    end

    local count = 0

    for line in string.gmatch(content, "[^\r\n]+") do

        local jobId = line:match("^([^|]+)")

        if jobId and jobId ~= "" then

            if not blockedJobs[jobId] then

                blockedJobs[jobId] = true
                count += 1

            end
        end
    end

    print(
        "[AntiAFK] โหลด Blocked JobId:",
        count
    )
end

loadJobFile()

-- =========================================================
-- SAVE JOB FILE
-- =========================================================

local function saveJobFile()

    if not hasFileAPI then
        return false
    end

    local jobs = {}

    for jobId in pairs(blockedJobs) do
        table.insert(jobs, jobId)
    end

    while #jobs > MAX_BLOCKED_JOBS do
        table.remove(jobs, 1)
    end

    local output = ""

    for _, jobId in ipairs(jobs) do
        output = output .. jobId .. "\n"
    end

    local success, err = pcall(function()

        writefile(
            JOB_FILE,
            output
        )

    end)

    if not success then

        warn(
            "[AntiAFK] บันทึก JobId ไม่สำเร็จ:",
            err
        )

        return false
    end

    return true
end

-- =========================================================
-- ADD BLOCKED JOB
-- =========================================================

local function addBlockedJob(jobId)

    if not jobId or jobId == "" then
        return false
    end

    if blockedJobs[jobId] then
        return true
    end

    blockedJobs[jobId] = true

    print("================================")
    print("[AntiAFK] 🚫 เพิ่ม Blocked JobId")
    print("[AntiAFK] JobId:", jobId)
    print("================================")

    local jobs = {}

    for id in pairs(blockedJobs) do
        table.insert(jobs, id)
    end

    while #jobs > MAX_BLOCKED_JOBS do

        local removeId =
            table.remove(jobs, 1)

        if removeId then
            blockedJobs[removeId] = nil
        end
    end

    saveJobFile()

    return true
end

-- =========================================================
-- CHECK BLOCKED JOB
-- =========================================================

local function isBlockedJob(jobId)

    if not jobId or jobId == "" then
        return false
    end

    return blockedJobs[jobId] == true
end

-- =========================================================
-- COUNT BLOCKED
-- =========================================================

local function getBlockedJobCount()

    local count = 0

    for _ in pairs(blockedJobs) do
        count += 1
    end

    return count
end

-- =========================================================
-- SAVE SERVER POOL
-- =========================================================

local function saveServerPool()

    if not hasFileAPI then
        return false
    end

    local output = ""

    local count = 0

    for jobId, info in pairs(serverPool) do

        if count >= SERVER_POOL_MAX then
            break
        end

        if jobId
            and jobId ~= ""
            and not isBlockedJob(jobId)
        then

            local playing =
                tonumber(info.playing) or 0

            local maxPlayers =
                tonumber(info.maxPlayers) or 0

            output =
                output
                .. jobId
                .. "|"
                .. playing
                .. "|"
                .. maxPlayers
                .. "\n"

            count += 1
        end
    end

    local success, err = pcall(function()

        writefile(
            SERVER_POOL_FILE,
            output
        )

    end)

    if not success then

        warn(
            "[AntiAFK] บันทึก Server Pool ไม่สำเร็จ:",
            err
        )

        return false
    end

    print(
        "[AntiAFK] 💾 บันทึก Server Pool:",
        count,
        "Server"
    )

    return true
end

    )

-- =========================================================
-- ADD TO POOL
-- =========================================================

local function addToServerPool(
    jobId,
    playing,
    maxPlayers
)

    if not jobId
        or jobId == ""
    then
        return
    end

    if isBlockedJob(jobId) then
        return
    end

    if jobId == game.JobId then
        return
    end

    serverPool[jobId] = {

        playing =
            tonumber(playing) or 0,

        maxPlayers =
            tonumber(maxPlayers) or 0
    }
end

-- =========================================================
-- REFRESH SERVER POOL
-- =========================================================

local function refreshServerPool()

    local placeId = game.PlaceId

    print("================================")
    print("[AntiAFK] 📥 กำลังโหลด Server Pool")
    print("[AntiAFK] Page Limit:", SERVER_PAGE_LIMIT)
    print("================================")

    local cursor = nil

    local added = 0

    for page = 1, SERVER_PAGE_LIMIT do

        local url =
            "https://games.roblox.com/v1/games/"
            .. tostring(placeId)
            .. "/servers/Public?sortOrder=Asc&limit=100"

        if cursor then

            url =
                url
                .. "&cursor="
                .. HttpService:UrlEncode(
                    cursor
                )

        end

        local success, result =
            pcall(function()

                return game:HttpGet(url)

            end)

        if not success then

            warn(
                "[AntiAFK] ❌ โหลด Server Page ไม่สำเร็จ:",
                page
            )

            break
        end

        local decodeSuccess, data =
            pcall(function()

                return HttpService:JSONDecode(
                    result
                )

            end)

        if not decodeSuccess
            or not data
            or not data.data
        then

            warn(
                "[AntiAFK] ❌ Server Data ไม่ถูกต้อง"
            )

            break
        end

        print(
            "[AntiAFK] Page",
            page,
            "| พบ",
            #data.data,
            "Server"
        )

        for _, server in ipairs(data.data) do

            local jobId =
                server.id

            local playing =
                tonumber(server.playing) or 0

            local maxPlayers =
                tonumber(server.maxPlayers) or 0

            if jobId
                and jobId ~= ""
                and jobId ~= game.JobId
                and not isBlockedJob(jobId)
            then

                local freeSlots =
                    maxPlayers - playing

                if maxPlayers > SERVER_SLOT_BUFFER
                    and freeSlots > SERVER_SLOT_BUFFER
                then

                    if not serverPool[jobId] then
                        added += 1
                    end

                    addToServerPool(
                        jobId,
                        playing,
                        maxPlayers
                    )
                end
            end
        end

        cursor =
            data.nextPageCursor

        if not cursor then
            break
        end
    end

    -- =====================================================
    -- LIMIT POOL
    -- =====================================================

    local jobs = {}

    for jobId in pairs(serverPool) do
        table.insert(
            jobs,
            jobId
        )
    end

    while #jobs > SERVER_POOL_MAX do

        local index =
            math.random(
                1,
                #jobs
            )

        local removeId =
            table.remove(
                jobs,
                index
            )

        if removeId then
            serverPool[removeId] = nil
        end
    end

    saveServerPool()

    local total = 0

    for _ in pairs(serverPool) do
        total += 1
    end

    print(
        "[AntiAFK] ✅ Server Pool พร้อม:",
        total,
        "Server"
    )

    print(
        "[AntiAFK] เพิ่มใหม่:",
        added
    )

    return total > 0
end

-- =========================================================
-- GET RANDOM SERVER FROM POOL
-- =========================================================

local function getRandomServerFromPool()

    local candidates = {}

    for jobId, info in pairs(serverPool) do

        if jobId
            and jobId ~= ""
            and jobId ~= game.JobId
            and not isBlockedJob(jobId)
        then

            local playing =
                tonumber(info.playing) or 0

            local maxPlayers =
                tonumber(info.maxPlayers) or 0

            local freeSlots =
                maxPlayers - playing

            if maxPlayers > SERVER_SLOT_BUFFER
                and freeSlots > SERVER_SLOT_BUFFER
            then

                table.insert(
                    candidates,
                    {
                        JobId = jobId,
                        Playing = playing,
                        MaxPlayers = maxPlayers
                    }
                )
            end
        end
    end

    if #candidates == 0 then
        return nil
    end

    return candidates[
        math.random(
            1,
            #candidates
        )
    ]
end

local function serverHop()

    if getgenv().AntiAFKHopping then

        print(
            "[AntiAFK] Server Hop กำลังทำงานอยู่แล้ว"
        )

        return
    end

    getgenv().AntiAFKHopping = true
    getgenv().AntiAFKRunning = false

    local placeId =
        game.PlaceId

    print("================================")
    print("[AntiAFK] 🔄 SERVER HOP")
    print(
        "[AntiAFK] Current JobId:",
        game.JobId
    )
    print(
        "[AntiAFK] Blocked Jobs:",
        getBlockedJobCount()
    )
    print("================================")

    -- =====================================================
    -- HOP LOOP
    -- =====================================================

    while true do

        local server =
            getRandomServerFromPool()

        -- =================================================
        -- POOL EMPTY
        -- =================================================

        if not server then

            warn(
                "[AntiAFK] ⚠️ Server Pool หมด"
            )

            print(
                "[AntiAFK] 📥 โหลด Server Pool ใหม่..."
            )

            local loaded =
                refreshServerPool()

            if not loaded then

                warn(
                    "[AntiAFK] ❌ โหลด Server ใหม่ไม่สำเร็จ"
                )

                task.wait(
                    HOP_RETRY_DELAY
                )

                continue
            end

            server =
                getRandomServerFromPool()

            if not server then

                task.wait(
                    HOP_RETRY_DELAY
                )

                continue
            end
        end

        local serverJobId =
            server.JobId

        -- =================================================
        -- REMOVE FROM POOL BEFORE TELEPORT
        -- =================================================

        removeFromServerPool(
            serverJobId
        )

        saveServerPool()

        print("================================")
        print(
            "[AntiAFK] 🎲 เลือก Server แบบสุ่ม"
        )
        print(
            "[AntiAFK] JobId:",
            serverJobId
        )
        print(
            "[AntiAFK] Players:",
            server.Playing
            .. "/"
            .. server.MaxPlayers
        )
        print(
            "[AntiAFK] Pool ที่เหลือ:"
        )

        local poolCount = 0

        for _ in pairs(serverPool) do
            poolCount += 1
        end

        print(
            "[AntiAFK]",
            poolCount,
            "Server"
        )

        print(
            "[AntiAFK] 🚀 กำลัง Teleport..."
        )

        print("================================")

        -- =================================================
        -- TELEPORT
        -- =================================================

        local teleportFailed = false
        local teleportResult = nil
        local teleportMessage = nil

        local connection

        connection =
            TeleportService.TeleportInitFailed:
            Connect(
                function(
                    player,
                    result,
                    message
                )

                    if player ~= Player then
                        return
                    end

                    teleportFailed = true
                    teleportResult = result
                    teleportMessage = message

                end
            )

        local callSuccess,
            callError =
            pcall(function()

                TeleportService:
                    TeleportToPlaceInstance(
                        placeId,
                        serverJobId,
                        Player
                    )

            end)

        if not callSuccess then

            warn(
                "[AntiAFK] ❌ Teleport Call Error:",
                callError
            )

            if connection then
                connection:Disconnect()
            end

            task.wait(
                HOP_RETRY_DELAY
            )

            continue
        end

        -- =================================================
        -- WAIT TELEPORT RESULT
        -- =================================================

        local timeout = 0

        while timeout < 10 do

            task.wait(0.25)

            timeout += 0.25

            if teleportFailed then
                break
            end
        end

        if connection then
            connection:Disconnect()
        end

        -- =================================================
        -- TELEPORT FAILED
        -- =================================================

        if teleportFailed then

            warn("================================")
            warn(
                "[AntiAFK] ❌ Teleport ไม่สำเร็จ"
            )

            warn(
                "[AntiAFK] JobId:",
                serverJobId
            )

            warn(
                "[AntiAFK] Result:",
                tostring(
                    teleportResult
                )
            )

            warn(
                "[AntiAFK] Message:",
                tostring(
                    teleportMessage
                )
            )

            warn(
                "[AntiAFK] 🗑️ ลบ JobId นี้ออกจาก Pool"
            )

            warn(
                "[AntiAFK] 🔄 ลอง Server ถัดไป..."
            )

            warn("================================")

            -- ไม่บันทึก Job นี้กลับเข้า Pool
            -- แล้วสุ่มตัวใหม่ทันที

            task.wait(0.5)

        else

            -- =================================================
            -- TELEPORT SENT
            -- =================================================

            print(
                "[AntiAFK] 📡 ส่ง Teleport แล้ว"
            )

            print(
                "[AntiAFK] ⏳ รอเข้า Server ใหม่..."
            )

            -- ไม่ยิง API ซ้ำ
            -- ถ้าเข้าได้ Script จะเริ่มใหม่
            -- ถ้า JobId ใหม่ถูก Blocked
            -- CURRENT JOB CHECK จะเรียก Hop อีกครั้ง

            return
        end
    end
end

local samePositionCount = 0

local function getPosition()

    local currentCharacter =
        Player.Character

    if not currentCharacter then
        return nil
    end

    local currentRoot =
        currentCharacter:
        FindFirstChild(
            "HumanoidRootPart"
        )

    if not currentRoot then
        return nil
    end

    return currentRoot.Position
end

-- =========================================================
-- POSITION CHECK
-- =========================================================

local function isSamePosition(
    pos1,
    pos2
)

    if not pos1 or not pos2 then
        return false
    end

    local distance =
        (pos1 - pos2).Magnitude

    print(
        "[AntiAFK] ระยะที่ขยับ:",
        math.floor(
            distance * 100
        ) / 100,
        "studs"
    )

    return distance <=
        POSITION_TOLERANCE
end
