-- =========================================================
-- LOAD BLOCKED JOB FILE
-- =========================================================

local function loadJobFile()

    if not hasFileAPI then
        return
    end

    if not isfile(JOB_FILE) then
        print("[AntiAFK] ยังไม่มี", JOB_FILE)
        return
    end

    local success, content = pcall(function()
        return readfile(JOB_FILE)
    end)

    if not success then
        warn("[AntiAFK] อ่าน Job File ไม่สำเร็จ:", content)
        return
    end

    if not content or content == "" then
        return
    end

    local count = 0

    for line in string.gmatch(content, "[^\r\n]+") do

        local jobId = line:match("^([^|]+)")

        if jobId then
            jobId = jobId:match("^%s*(.-)%s*$")
        end

        if jobId and jobId ~= "" then

            if not blockedJobs[jobId] then
                blockedJobs[jobId] = true
                count += 1
            end

        end
    end

    print("[AntiAFK] โหลด Blocked JobId:", count)
end

loadJobFile()

-- =========================================================
-- CHECK BLOCKED JOB
-- =========================================================

local function isBlockedJob(jobId)

    if not jobId or jobId == "" then
        return false
    end

    return blockedJobs[jobId] == true
end

-- =========================================================
-- COUNT BLOCKED
-- =========================================================

local function getBlockedJobCount()

    local count = 0

    for _ in pairs(blockedJobs) do
        count += 1
    end

    return count
end

-- =========================================================
-- SAVE BLOCKED JOB FILE
-- =========================================================

local function saveJobFile()

    if not hasFileAPI then
        return false
    end

    local jobs = {}

    for jobId in pairs(blockedJobs) do
        table.insert(jobs, jobId)
    end

    while #jobs > MAX_BLOCKED_JOBS do

        local removeId = table.remove(jobs, 1)

        if removeId then
            blockedJobs[removeId] = nil
        end
    end

    local output = ""

    for _, jobId in ipairs(jobs) do
        output = output .. jobId .. "\n"
    end

    local success, err = pcall(function()
        writefile(JOB_FILE, output)
    end)

    if not success then
        warn(
            "[AntiAFK] บันทึก Blocked Job ไม่สำเร็จ:",
            err
        )
        return false
    end

    return true
end

-- =========================================================
-- ADD BLOCKED JOB
-- =========================================================

local function addBlockedJob(jobId)

    if not jobId or jobId == "" then
        return false
    end

    if blockedJobs[jobId] then

        print(
            "[AntiAFK] ⚠️ JobId นี้ Blocked อยู่แล้ว:",
            jobId
        )

        return true
    end

    blockedJobs[jobId] = true

    print("================================")
    print("[AntiAFK] 🚫 เพิ่ม Blocked JobId")
    print("[AntiAFK] JobId:", jobId)
    print("================================")

    saveJobFile()

    return true
end

-- =========================================================
-- LOAD SERVER CACHE
-- =========================================================

local function loadServerCache()

    serverCache = {}

    if not hasFileAPI then
        warn("[AntiAFK] File API ไม่มี -> โหลด Cache ไม่ได้")
        return false
    end

    if not isfile(SERVER_CACHE_FILE) then
        print("[AntiAFK] ยังไม่มี", SERVER_CACHE_FILE)
        return false
    end

    local success, content = pcall(function()
        return readfile(SERVER_CACHE_FILE)
    end)

    if not success then

        warn(
            "[AntiAFK] อ่าน Server Cache ไม่สำเร็จ:",
            content
        )

        return false
    end

    if not content or content == "" then

        print("[AntiAFK] server_cache.txt ว่าง")

        return false
    end

    local now = os.time()

    local count = 0
    local duplicateCount = 0
    local blockedCount = 0
    local expiredCount = 0

    for line in string.gmatch(
        content,
        "[^\r\n]+"
    ) do

        -- =================================================
        -- FORMAT
        --
        -- JobId|Players|MaxPlayers|Timestamp
        -- =================================================

        local jobId,
            playerCount,
            maxPlayers,
            timestamp =
            line:match(
                "^([^|]+)|([^|]+)|([^|]+)|([^|]+)"
            )

        if jobId then
            jobId =
                jobId:match("^%s*(.-)%s*$")
        end

        playerCount =
            tonumber(playerCount) or 0

        maxPlayers =
            tonumber(maxPlayers) or 0

        timestamp =
            tonumber(timestamp) or 0

        if jobId and jobId ~= "" then

            -- =================================================
            -- BLOCKED
            -- =================================================

            if isBlockedJob(jobId) then

                blockedCount += 1

            -- =================================================
            -- EXPIRED
            -- =================================================

            elseif timestamp <= 0
                or now - timestamp >= SERVER_CACHE_EXPIRE_TIME
            then

                expiredCount += 1

                print(
                    "[AntiAFK] ⏰ Cache หมดอายุ:",
                    jobId
                )

            -- =================================================
            -- DUPLICATE
            -- =================================================

            elseif serverCache[jobId] then

                duplicateCount += 1

                -- ถ้ามีซ้ำ ให้เลือกข้อมูลที่ Timestamp ใหม่กว่า

                if timestamp >
                    serverCache[jobId].timestamp
                then

                    serverCache[jobId] = {
                        players = playerCount,
                        maxPlayers = maxPlayers,
                        timestamp = timestamp
                    }
                end

            -- =================================================
            -- NEW
            -- =================================================

            else

                serverCache[jobId] = {
                    players = playerCount,
                    maxPlayers = maxPlayers,
                    timestamp = timestamp
                }

                count += 1
            end
        end
    end

    print("================================")
    print("[AntiAFK] 📦 โหลด Server Cache")
    print("[AntiAFK] Valid:", count)
    print("[AntiAFK] Duplicate:", duplicateCount)
    print("[AntiAFK] Blocked:", blockedCount)
    print("[AntiAFK] Expired:", expiredCount)
    print("================================")

    return count > 0
end

-- =========================================================
-- SAVE SERVER CACHE
-- =========================================================

local function saveServerCache()

    if not hasFileAPI then
        return false
    end

    local now = os.time()

    local output = ""

    local count = 0

    local written = {}

    for jobId, data in pairs(serverCache) do

        if not written[jobId]
            and not isBlockedJob(jobId)
            and data
        then

            local timestamp =
                tonumber(data.timestamp) or 0

            -- =================================================
            -- ไม่เขียนข้อมูลที่หมดอายุ
            -- =================================================

            if timestamp > 0
                and now - timestamp < SERVER_CACHE_EXPIRE_TIME
            then

                written[jobId] = true

                output =
                    output
                    .. jobId
                    .. "|"
                    .. tostring(
                        data.players or 0
                    )
                    .. "|"
                    .. tostring(
                        data.maxPlayers or 0
                    )
                    .. "|"
                    .. tostring(timestamp)
                    .. "\n"

                count += 1
            end
        end
    end

    local success, err =
        pcall(function()

            writefile(
                SERVER_CACHE_FILE,
                output
            )

        end)

    if not success then

        warn(
            "[AntiAFK] ❌ บันทึก Server Cache ไม่สำเร็จ:",
            err
        )

        return false
    end

    print(
        "[AntiAFK] 💾 Server Cache:",
        count,
        "JobId"
    )

    return true
end

-- =========================================================
-- REMOVE EXPIRED CACHE
-- =========================================================

local function removeExpiredCache()

    local now = os.time()

    local removedExpired = 0
    local removedBlocked = 0

    for jobId, data in pairs(serverCache) do

        if isBlockedJob(jobId) then

            serverCache[jobId] = nil
            removedBlocked += 1

        else

            local timestamp =
                tonumber(
                    data.timestamp
                ) or 0

            if timestamp <= 0
                or now - timestamp >= SERVER_CACHE_EXPIRE_TIME
            then

                serverCache[jobId] = nil
                removedExpired += 1
            end
        end
    end

    if removedExpired > 0 then

        print(
            "[AntiAFK] ⏰ ลบ Cache หมดอายุ:",
            removedExpired
        )
    end

    if removedBlocked > 0 then

        print(
            "[AntiAFK] 🚫 ลบ Blocked จาก Cache:",
            removedBlocked
        )
    end

    if removedExpired > 0
        or removedBlocked > 0
    then

        saveServerCache()
    end
end

-- UPDATE CURRENT SERVER
-- =========================================================

local function updateCurrentServerCache()

    local currentJobId =
        game.JobId

    if not currentJobId
        or currentJobId == ""
    then
        return false
    end

    -- =================================================
    -- Blocked ห้ามบันทึก
    -- =================================================

    if isBlockedJob(currentJobId) then

        if serverCache[currentJobId] then

            serverCache[currentJobId] = nil

            saveServerCache()
        end

        return false
    end

    -- =================================================
    -- จำนวนผู้เล่น
    -- =================================================

    local playerCount =
        #Players:GetPlayers()

    local maxPlayers =
        Players.MaxPlayers or 0

    local timestamp =
        os.time()

    local existed =
        serverCache[currentJobId] ~= nil

    -- =================================================
    -- UPDATE / INSERT
    -- =================================================

    serverCache[currentJobId] = {

        players = playerCount,

        maxPlayers = maxPlayers,

        timestamp = timestamp
    }

    print("================================")

    if existed then

        print(
            "[AntiAFK] 🔄 UPDATE Server"
        )

    else

        print(
            "[AntiAFK] ➕ ADD Server"
        )
    end

    print(
        "[AntiAFK] JobId:",
        currentJobId
    )

    print(
        "[AntiAFK] Players:",
        playerCount
        .. "/"
        .. maxPlayers
    )

    print(
        "[AntiAFK] Timestamp:",
        timestamp
    )

    print(
        "[AntiAFK] Cache Age: 0 นาที"
    )

    print("================================")

    saveServerCache()

    return true
end

-- =========================================================
-- GET VALID CACHED JOBS
-- =========================================================

local function getValidCachedJobs()

    local now =
        os.time()

    local currentJobId =
        game.JobId

    local candidates = {}

    for jobId, data in pairs(serverCache) do

        if jobId
            and jobId ~= ""
            and jobId ~= currentJobId
            and not isBlockedJob(jobId)
        then

            local players =
                tonumber(
                    data.players
                ) or 0

            local maxPlayers =
                tonumber(
                    data.maxPlayers
                ) or 0

            local timestamp =
                tonumber(
                    data.timestamp
                ) or 0

            -- =================================================
            -- ต้องมี Timestamp
            -- =================================================

            if timestamp > 0 then

                local age =
                    now - timestamp

                -- =================================================
                -- ต้องไม่เกิน 1 ชั่วโมง
                -- =================================================

                if age <
                    SERVER_CACHE_EXPIRE_TIME
                then

                    -- =================================================
                    -- ต้องไม่เต็ม
                    -- =================================================

                    if maxPlayers <= 0
                        or players < maxPlayers
                    then

                        table.insert(
                            candidates,
                            {
                                jobId = jobId,

                                players = players,

                                maxPlayers =
                                    maxPlayers,

                                timestamp =
                                    timestamp,

                                age = age
                            }
                        )
                    end
                end
            end
        end
    end

    return candidates
end

-- =========================================================
-- SHUFFLE
-- =========================================================

math.randomseed(
    os.time()
    + tonumber(
        string.sub(
            tostring(
                tick()
            ),
            -6
        )
    )
)

local function shuffle(list)

    for i = #list, 2, -1 do

        local j =
            math.random(
                1,
                i
            )

        list[i],
        list[j] =
            list[j],
            list[i]
    end

    return list
end

local function serverHop()

    if getgenv().AntiAFKHopping then

        print(
            "[AntiAFK] Server Hop กำลังทำงานอยู่แล้ว"
        )

        return false
    end

    getgenv().AntiAFKHopping = true
    getgenv().AntiAFKRunning = false

    print("================================")
    print("[AntiAFK] 🔄 SERVER CACHE HOP")
    print("[AntiAFK] Current JobId:", game.JobId)
    print(
        "[AntiAFK] Blocked:",
        getBlockedJobCount()
    )
    print(
        "[AntiAFK] Cache:",
        getServerCacheCount()
    )
    print("================================")

    -- =================================================
    -- โหลด Cache ใหม่
    -- =================================================

    loadServerCache()

    -- =================================================
    -- ลบ Expired + Blocked
    -- =================================================

    removeExpiredCache()

    -- =================================================
    -- หา Server ที่ยังใช้ได้
    -- =================================================

    local validJobs =
        getValidCachedJobs()

    if #validJobs == 0 then

        warn("================================")
        warn(
            "[AntiAFK] ❌ ไม่มี Server ที่ใช้ได้"
        )
        warn(
            "[AntiAFK] Cache:",
            getServerCacheCount()
        )
        warn(
            "[AntiAFK] Blocked:",
            getBlockedJobCount()
        )
        warn("================================")

        getgenv().AntiAFKHopping =
            false

        return false
    end

    -- =================================================
    -- สุ่ม
    -- =================================================

    shuffle(validJobs)

    print(
        "[AntiAFK] 🎲 Server ที่ใช้ได้:",
        #validJobs
    )

    -- =================================================
    -- ลอง Server ที่สุ่มได้
    -- =================================================

    for index, serverInfo in ipairs(validJobs) do

        local serverJobId =
            serverInfo.jobId

        print("================================")

        print(
            "[AntiAFK] 🎯 Random Server:",
            index
            .. "/"
            .. #validJobs
        )

        print(
            "[AntiAFK] JobId:",
            serverJobId
        )

        print(
            "[AntiAFK] Players:",
            serverInfo.players
            .. "/"
            .. serverInfo.maxPlayers
        )

        print(
            "[AntiAFK] Cache Age:",
            math.floor(
                serverInfo.age / 60
            ),
            "นาที"
        )

        print("================================")

        local teleportFailed =
            false

        local teleportResult =
            nil

        local teleportMessage =
            nil

        local connection

        connection =
            TeleportService
            .TeleportInitFailed
            :Connect(
                function(
                    player,
                    result,
                    message
                )

                    if player ~= Player then
                        return
                    end

                    teleportFailed =
                        true

                    teleportResult =
                        result

                    teleportMessage =
                        message
                end
            )

        local callSuccess,
            callError =
            pcall(function()

                TeleportService:
                    TeleportToPlaceInstance(
                        game.PlaceId,
                        serverJobId,
                        Player
                    )
            end)

        if not callSuccess then

            warn(
                "[AntiAFK] ❌ Teleport Error:",
                callError
            )

            if connection then
                connection:Disconnect()
            end

            task.wait(
                HOP_RETRY_DELAY
            )

            continue
        end

        -- =================================================
        -- รอผล Teleport
        -- =================================================

        local timeout =
            0

        while timeout < TELEPORT_TIMEOUT do

            task.wait(0.25)

            timeout += 0.25

            if teleportFailed then
                break
            end
        end

        if connection then
            connection:Disconnect()
        end

        -- =================================================
        -- FAILED
        -- =================================================

        if teleportFailed then

            warn("================================")

            warn(
                "[AntiAFK] ❌ เข้า Server ไม่สำเร็จ"
            )

            warn(
                "[AntiAFK] JobId:",
                serverJobId
            )

            warn(
                "[AntiAFK] Result:",
                tostring(
                    teleportResult
                )
            )

            warn(
                "[AntiAFK] Message:",
                tostring(
                    teleportMessage
                )
            )

            warn(
                "[AntiAFK] 🔄 สุ่มตัวถัดไป"
            )

            warn("================================")

            task.wait(
                HOP_RETRY_DELAY
            )

            continue
        end

        -- =================================================
        -- SUCCESS
        -- =================================================

        print("================================")

        print(
            "[AntiAFK] 📡 ส่ง Teleport แล้ว"
        )

        print(
            "[AntiAFK] JobId:",
            serverJobId
        )

        print(
            "[AntiAFK] Players:",
            serverInfo.players
            .. "/"
            .. serverInfo.maxPlayers
        )

        print(
            "[AntiAFK] ⏳ รอเข้า Server..."
        )

        print("================================")

        return true
    end

    warn(
        "[AntiAFK] ❌ ไม่มี Server ที่ Teleport สำเร็จ"
    )

    getgenv().AntiAFKHopping =
        false

    return false
end

-- =========================================================

-- =========================================================

local function isSamePosition(
    pos1,
    pos2
)

    if not pos1 or not pos2 then
        return false
    end

    local distance =
        (pos1 - pos2).Magnitude

    print(
        "[AntiAFK] ระยะที่ขยับ:",
        math.floor(
            distance * 100
        ) / 100,
        "studs"
    )

    return distance <=
        POSITION_TOLERANCE
end

                    getBlockedJobCount()
                )
                .. " | Cache: "
                .. tostring(
                    getServerCacheCount()
                )
            )

        elseif getgenv().AntiAFKHopping then

            print(
                "[AntiAFK] STATUS: HOPPING"
                .. " | JobId: "
                .. tostring(
                    game.JobId
                )
                .. " | Cache: "
                .. tostring(
                    getServerCacheCount()
                )
            )

        else

        end
    end
end)

-- =========================================================
-- MAIN LOOP
-- =========================================================

print(
    "[AntiAFK] Main Loop เริ่มทำงานแล้ว"
)

while getgenv().AntiAFKRunning do

    task.wait(
        CHECK_INTERVAL
    )

    if not getgenv().AntiAFKRunning then
        break
    end

    local currentPosition =
        getPosition()

    if not currentPosition then

        warn(
            "[AntiAFK] ไม่พบ Character/HumanoidRootPart"
        )

        continue
    end

    -- =====================================================
    -- FIRST CHECK
    -- =====================================================

    if not lastPosition then

        lastPosition =
            currentPosition

        samePositionCount =
            0

        print(
            "[AntiAFK] บันทึกตำแหน่งเริ่มต้นแล้ว"
        )

        print(
            "[AntiAFK] Check: 0/"
            .. REQUIRED_CHECKS
        )

    -- =====================================================
    -- SAME POSITION
    -- =====================================================

    elseif isSamePosition(
        lastPosition,
        currentPosition
    ) then

        samePositionCount += 1

        print(
            "[AntiAFK] อยู่จุดเดิม | "
            .. samePositionCount
            .. "/"
            .. REQUIRED_CHECKS
        )

        -- =================================================
        -- AFK TRIGGER
        -- =================================================

        if samePositionCount >=
            REQUIRED_CHECKS
        then

            print("================================")

            print(
                "[AntiAFK] ⚠️ AFK TRIGGERED"
            )

            local currentJobId =
                game.JobId

            print(
                "[AntiAFK] นิ่งครบ "
                .. REQUIRED_CHECKS
                .. " ครั้ง"
            )

            -- =================================================
            -- ADD BLOCKED
            -- =================================================

            addBlockedJob(
                currentJobId
            )

            -- =================================================
            -- REMOVE FROM CACHE
            -- =================================================

            if serverCache[currentJobId] then

                serverCache[currentJobId] =
                    nil

                print(
                    "[AntiAFK] 🧹 ลบ JobId จาก Server Cache"
                )

                saveServerCache()
            end

            print(
                "[AntiAFK] 💾 บันทึก Blocked JobId แล้ว"
            )

            print(
                "[AntiAFK] 🔴 สาเหตุ: AFK"
            )

            print(
                "[AntiAFK] 🔴 กำลังเตะออก..."
            )

            print("================================")

            getgenv().AntiAFKRunning =
                false

            task.wait(1)

            Player:Kick(
                "AntiAFK: AFK Server"
            )

            break
        end

    -- =====================================================
    -- POSITION CHANGED
    -- =====================================================

    else

        samePositionCount =
            0

        print(
            "[AntiAFK] ตำแหน่งเปลี่ยน"
            .. " -> Reset 0/"
            .. REQUIRED_CHECKS
        )
    end

    lastPosition =
        currentPosition
end
