-- =========================================================
-- LOAD JOB FILE
-- =========================================================

local function loadJobFile()

    if not hasFileAPI then
        return
    end

    if not isfile(JOB_FILE) then
        print("[AntiAFK] ยังไม่มีไฟล์", JOB_FILE)
        return
    end

    local success, content = pcall(function()
        return readfile(JOB_FILE)
    end)

    if not success then
        warn("[AntiAFK] อ่านไฟล์ JobId ไม่สำเร็จ:", content)
        return
    end

    if not content or content == "" then
        print("[AntiAFK] ไฟล์ JobId ว่าง")
        return
    end

    local count = 0

    for line in string.gmatch(content, "[^\r\n]+") do

        local jobId = line:match("^([^|]+)")

        if jobId and jobId ~= "" then

            if not blockedJobs[jobId] then
                blockedJobs[jobId] = true
                count += 1
            end

        end
    end

    print(
        "[AntiAFK] โหลด JobId จากไฟล์แล้ว:",
        count,
        "รายการ"
    )
end

loadJobFile()

-- =========================================================
-- SAVE JOB FILE
-- =========================================================

local function saveJobFile()

    if not hasFileAPI then
        warn(
            "[AntiAFK] ไม่มี File API -> "
            .. "ไม่สามารถบันทึกไฟล์ได้"
        )

        return false
    end

    local jobs = {}

    for jobId in pairs(blockedJobs) do
        table.insert(jobs, jobId)
    end

    -- จำกัดจำนวน JobId
    while #jobs > MAX_BLOCKED_JOBS do
        table.remove(jobs, 1)
    end

    local output = ""

    for _, jobId in ipairs(jobs) do
        output = output .. jobId .. "\n"
    end

    local success, err = pcall(function()
        writefile(JOB_FILE, output)
    end)

    if not success then

        warn(
            "[AntiAFK] บันทึก JobId ไม่สำเร็จ:",
            err
        )

        return false
    end

    print(
        "[AntiAFK] บันทึก JobId ลง",
        JOB_FILE,
        "แล้ว"
    )

    return true
end

-- =========================================================
-- ADD BLOCKED JOB
-- =========================================================

local function addBlockedJob(jobId)

    if not jobId or jobId == "" then

        warn(
            "[AntiAFK] JobId ว่าง -> ไม่บันทึก"
        )

        return false
    end

    if blockedJobs[jobId] then

        print(
            "[AntiAFK] JobId นี้มีอยู่แล้ว:",
            jobId
        )

        return true
    end

    blockedJobs[jobId] = true

    print("================================")
    print("[AntiAFK] เพิ่ม JobId ที่ถูก Kick")
    print("[AntiAFK] JobId:", jobId)
    print("================================")

    -- จำกัดจำนวน JobId
    local jobs = {}

    for id in pairs(blockedJobs) do
        table.insert(jobs, id)
    end

    while #jobs > MAX_BLOCKED_JOBS do

        local removeId = table.remove(jobs, 1)

        if removeId then
            blockedJobs[removeId] = nil
        end
    end

    return saveJobFile()
end

-- =========================================================
-- CHECK BLOCKED JOB
-- =========================================================

local function isBlockedJob(jobId)

    if not jobId or jobId == "" then
        return false
    end

    return blockedJobs[jobId] == true
end

-- =========================================================
-- COUNT BLOCKED JOBS
-- =========================================================

local function getBlockedJobCount()

    local count = 0

    for _ in pairs(blockedJobs) do
        count += 1
    end

    return count
end

-- =========================================================
-- PRINT JOB LIST
-- =========================================================

local function printBlockedJobs()

    local count = 0

    print("[AntiAFK] ===== Blocked JobIds =====")

    for jobId in pairs(blockedJobs) do

        count += 1

        print(
            "[AntiAFK]",

        )
    end

    print(
        "[AntiAFK] ทั้งหมด:",
        count,
        "JobId"
    )

    print("[AntiAFK] =========================")
end

-- =========================================================
-- SERVER HOP
-- =========================================================

local function serverHop()

    if getgenv().AntiAFKHopping then

        print(
            "[AntiAFK] Server Hop กำลังทำงานอยู่แล้ว"
        )

        return
    end

    getgenv().AntiAFKHopping = true
    getgenv().AntiAFKRunning = false

    local currentJobId = game.JobId
    local placeId = game.PlaceId

    print("================================")
    print("[AntiAFK] 🔄 SERVER HOP")
    print("[AntiAFK] Current JobId:", currentJobId)
    print("[AntiAFK] Blocked Jobs:", getBlockedJobCount())
    print("================================")

    local cursor = nil
    local triedServers = {}

    -- กันไม่ให้เลือก JobId ปัจจุบัน
    triedServers[currentJobId] = true

    for page = 1, SERVER_PAGE_LIMIT do

        print(
            "[AntiAFK] กำลังค้นหา Server..."
            .. " Page:",
            page
        )

        local url =
            "https://games.roblox.com/v1/games/"
            .. tostring(placeId)
            .. "/servers/Public?sortOrder=Asc&limit=100"

        if cursor then
            url = url
                .. "&cursor="
                .. HttpService:UrlEncode(cursor)
        end

        local success, result = pcall(function()

            return game:HttpGet(url)

        end)

        if not success then

            warn(
                "[AntiAFK] ดึง Server List ไม่สำเร็จ:",
                result
            )

            task.wait(HOP_RETRY_DELAY)

        else

            local decodeSuccess, data =
                pcall(function()

                    return HttpService:JSONDecode(result)

                end)

            if not decodeSuccess then

                warn(
                    "[AntiAFK] Decode Server List ไม่สำเร็จ:",
                    data
                )

                task.wait(HOP_RETRY_DELAY)

            elseif data and data.data then

                print(
                    "[AntiAFK] พบ Server:",
                    #data.data
                )

                for _, server in ipairs(data.data) do

                    local serverJobId = server.id
                    local playing = server.playing or 0
                    local maxPlayers = server.maxPlayers or 0

                    -- =================================================
                    -- SERVER VALIDATION
                    -- =================================================

                    if serverJobId
                        and serverJobId ~= ""
                        and serverJobId ~= currentJobId
                        and not triedServers[serverJobId]
                        and not isBlockedJob(serverJobId)
                        and playing < maxPlayers
                    then

                        print("================================")
                        print("[AntiAFK] ✅ พบ Server ใหม่")
                        print("[AntiAFK] JobId:", serverJobId)
                        print(
                            "[AntiAFK] Players:",
                            playing .. "/" .. maxPlayers
                        )
                        print("[AntiAFK] กำลัง Teleport...")
                        print("================================")

                        triedServers[serverJobId] = true

                        local teleportSuccess, teleportError =
                            pcall(function()

                                TeleportService:TeleportToPlaceInstance(
                                    placeId,
                                    serverJobId,
                                    Player
                                )

                            end)

                        if teleportSuccess then

                            print(
                                "[AntiAFK] Teleport ถูกส่งแล้ว"
                            )

                            return true

                        else

                            warn(
                                "[AntiAFK] Teleport ไม่สำเร็จ:",
                                teleportError
                            )

                        end
                    end
                end

                cursor = data.nextPageCursor

                if not cursor then

                    print(
                        "[AntiAFK] ไม่มีหน้า Server เพิ่มแล้ว"
                    )

                    break
                end

            end
        end
    end

    -- =========================================================
    -- NO SERVER FOUND
    -- =========================================================

    warn(
        "[AntiAFK] ❌ หา Server ใหม่ที่ไม่ Blocked ไม่พบ"
    )

    print(
        "[AntiAFK] รอ",
        HOP_RETRY_DELAY,
        "วินาที แล้วลองใหม่"
    )

    task.wait(HOP_RETRY_DELAY)

    getgenv().AntiAFKHopping = false
    getgenv().AntiAFKRunning = true

    -- ลอง Hop ใหม่
    task.spawn(serverHop)
end

-- =========================================================
-- POSITION
-- =========================================================

local samePositionCount = 0
local lastPosition = nil

local function getPosition()

    local character = Player.Character

    if not character then
        return nil
    end

    local root =
        character:FindFirstChild("HumanoidRootPart")

    if not root then
        return nil
    end

    return root.Position
end

-- =========================================================
-- POSITION CHECK
-- =========================================================

local function isSamePosition(pos1, pos2)

    if not pos1 or not pos2 then
        return false
    end

    local distance =
        (pos1 - pos2).Magnitude

    print(
        "[AntiAFK] ระยะที่ขยับ:",
        math.floor(distance * 100) / 100,
        "studs"
    )

    return distance <= POSITION_TOLERANCE
end

        if getgenv().AntiAFKRunning then

                .. tostring(game.JobId)
                .. " | Check: "

                .. " | Current JobId: "
                .. tostring(game.JobId)
                .. " | Blocked: "
                .. tostring(getBlockedJobCount())
            )

        else

            warn(
                "[AntiAFK] STATUS: STOPPED"
            )

            break
        end
    end
end)

-- =========================================================
-- MAIN LOOP
-- =========================================================

print("[AntiAFK] Main Loop เริ่มทำงานแล้ว")

while true do

    task.wait(CHECK_INTERVAL)

    if not getgenv().AntiAFKRunning then
        break
    end

    local currentPosition =
        getPosition()

    if not currentPosition then

        warn(
            "[AntiAFK] ไม่พบ Character/HumanoidRootPart"
        )

        continue
    end

    -- =====================================================
    -- FIRST CHECK
    -- =====================================================

    if not lastPosition then

        lastPosition = currentPosition
        samePositionCount = 0

        print(
            "[AntiAFK] บันทึกตำแหน่งเริ่มต้นแล้ว"
        )

        print(
            "[AntiAFK] Check: 0/"
            .. REQUIRED_CHECKS
        )

    -- =====================================================
    -- SAME POSITION
    -- =====================================================

    elseif isSamePosition(
        lastPosition,
        currentPosition
    ) then

        samePositionCount += 1

        print(
            "[AntiAFK] อยู่จุดเดิม | "
            .. samePositionCount
            .. "/"
            .. REQUIRED_CHECKS
        )

        -- =================================================
        -- AFK TRIGGER
        -- =================================================

        if samePositionCount >= REQUIRED_CHECKS then

            print("================================")
            print("[AntiAFK] ⚠️ AFK TRIGGERED")
            print(
                "[AntiAFK] นิ่งครบ "
                .. REQUIRED_CHECKS
                .. " ครั้ง"
            )

            -- =================================================
            -- SAVE CURRENT JOB ID
            -- =================================================

            local currentJobId = game.JobId

            print(
                "[AntiAFK] กำลังบันทึก JobId..."
            )

            local saved =
                addBlockedJob(currentJobId)

            if saved then

                print(
                    "[AntiAFK] ✅ JobId ถูกบันทึกแล้ว"
                )

                print(
                    "[AntiAFK] JobId:",
                    currentJobId
                )

            else

                warn(
                    "[AntiAFK] ❌ บันทึก JobId ไม่สำเร็จ"
                )

            end

            -- =================================================
            -- KICK ONLY
            -- =================================================

            print(
                "[AntiAFK] 🔴 สาเหตุ: AFK"
            )

            print(
                "[AntiAFK] กำลังเตะออกจาก Server..."
            )

            print("================================")

            getgenv().AntiAFKRunning = false

            task.wait(1)

            Player:Kick(
                "AntiAFK: AFK Server"
            )

            break
        end

    -- =====================================================
    -- POSITION CHANGED
    -- =====================================================

    else

        samePositionCount = 0

        print(
            "[AntiAFK] ตำแหน่งเปลี่ยน"
            .. " -> Reset 0/"
            .. REQUIRED_CHECKS
        )
    end

    lastPosition = currentPosition
end
