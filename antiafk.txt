    getgenv().AntiAFKBlockedJobs = {}
end

local blockedJobs = getgenv().AntiAFKBlockedJobs

-- =========================================================
-- LOAD JOB FILE
-- =========================================================

local function loadJobFile()

    if not hasFileAPI then
        return
    end

    if not isfile(JOB_FILE) then
        print("[AntiAFK] ยังไม่มีไฟล์", JOB_FILE)
        return
    end

    local success, content = pcall(function()
        return readfile(JOB_FILE)
    end)

    if not success then
        warn("[AntiAFK] อ่าน Job File ไม่สำเร็จ:", content)
        return
    end

    if not content or content == "" then
        return
    end

    local count = 0

    for line in string.gmatch(content, "[^\r\n]+") do

        local jobId = line:match("^([^|]+)")

        if jobId then
            jobId = jobId:match("^%s*(.-)%s*$")
        end

        if jobId and jobId ~= "" then

            if not blockedJobs[jobId] then
                blockedJobs[jobId] = true
                count += 1
            end

        end
    end

    print(
        "[AntiAFK] โหลด Blocked JobId:",
        count
    )
end

loadJobFile()

-- =========================================================
-- SAVE JOB FILE
-- =========================================================

local function saveJobFile()

    if not hasFileAPI then
        return false
    end

    local jobs = {}

    for jobId in pairs(blockedJobs) do
        table.insert(jobs, jobId)
    end

    while #jobs > MAX_BLOCKED_JOBS do

        local removeId =
            table.remove(jobs, 1)

        if removeId then
            blockedJobs[removeId] = nil
        end
    end

    local output = ""

    for _, jobId in ipairs(jobs) do
        output =
            output
            .. jobId
            .. "\n"
    end

    local success, err =
        pcall(function()

            writefile(
                JOB_FILE,
                output
            )

        end)

    if not success then

        warn(
            "[AntiAFK] บันทึก JobId ไม่สำเร็จ:",
            err
        )

        return false
    end

    return true
end

-- =========================================================
-- ADD BLOCKED JOB
-- =========================================================

local function addBlockedJob(jobId)

    if not jobId or jobId == "" then
        return false
    end

    -- มีอยู่แล้ว ไม่เพิ่มซ้ำ
    if blockedJobs[jobId] then
        return true
    end

    blockedJobs[jobId] = true

    print("================================")
    print("[AntiAFK] 🚫 เพิ่ม Blocked JobId")
    print("[AntiAFK] JobId:", jobId)
    print("================================")

    local jobs = {}

    for id in pairs(blockedJobs) do
        table.insert(jobs, id)
    end

    while #jobs > MAX_BLOCKED_JOBS do

        local removeId =
            table.remove(jobs, 1)

        if removeId then
            blockedJobs[removeId] = nil
        end
    end

    saveJobFile()

    return true
end

-- =========================================================
-- CHECK BLOCKED JOB
-- =========================================================

local function isBlockedJob(jobId)

    if not jobId or jobId == "" then
        return false
    end

    return blockedJobs[jobId] == true
end

-- =========================================================
-- COUNT BLOCKED
-- =========================================================

local function getBlockedJobCount()

    local count = 0

    for _ in pairs(blockedJobs) do
        count += 1
    end

    return count
end

local serverCache = {}

-- =========================================================
-- LOAD SERVER CACHE
-- =========================================================

local function loadServerCache()

    serverCache = {}

    if not hasFileAPI then

        warn(
            "[AntiAFK] File API ไม่มี -> อ่าน server_cache.txt ไม่ได้"
        )

        return false
    end

    if not isfile(SERVER_CACHE_FILE) then

        print(
            "[AntiAFK] ยังไม่มี",
            SERVER_CACHE_FILE
        )

        return false
    end

    local success, content =
        pcall(function()

            return readfile(
                SERVER_CACHE_FILE
            )

        end)

    if not success then

        warn(
            "[AntiAFK] อ่าน Server Cache ไม่สำเร็จ:",
            content
        )

        return false
    end

    if not content or content == "" then

        warn(
            "[AntiAFK] server_cache.txt ว่าง"
        )

        return false
    end

    local count = 0
    local seen = {}

    for line in string.gmatch(
        content,
        "[^\r\n]+"
    ) do

        local jobId =
            line:match("^([^|]+)")

        if jobId then
            jobId =
                jobId:match("^%s*(.-)%s*$")
        end

        if jobId
            and jobId ~= ""
            and not seen[jobId]
        then

            seen[jobId] = true

            table.insert(
                serverCache,
                jobId
            )

            count += 1
        end
    end

    print(
        "[AntiAFK] 📦 โหลด Server Cache:",
        count,
        "JobId"
    )

    return count > 0
end

-- =========================================================
-- SAVE CURRENT JOB TO SERVER CACHE
-- =========================================================

local function saveCurrentJobToCache()

    local jobId = game.JobId

    if not jobId or jobId == "" then

        warn(
            "[AntiAFK] ❌ ไม่มี JobId ปัจจุบัน"
        )

        return false
    end

    -- =====================================================
    -- BLOCKED CHECK
    -- =====================================================

    if isBlockedJob(jobId) then

        warn(
            "[AntiAFK] 🚫 JobId นี้เป็น Blocked"
        )

        warn(
            "[AntiAFK] ❌ ไม่บันทึกลง Server Cache:",
            jobId
        )

        return false
    end

    if not hasFileAPI then

        warn(
            "[AntiAFK] ❌ File API ไม่มี"
        )

        return false
    end

    -- =====================================================
    -- READ EXISTING CACHE
    -- =====================================================

    local content = ""

    if isfile(SERVER_CACHE_FILE) then

        local success, result =
            pcall(function()

                return readfile(
                    SERVER_CACHE_FILE
                )

            end)

        if success and result then
            content = result
        end
    end

    -- =====================================================
    -- CHECK DUPLICATE
    -- =====================================================

    local alreadyExists = false

    for line in string.gmatch(
        content,
        "[^\r\n]+"
    ) do

        local existingJobId =
            line:match("^([^|]+)")

        if existingJobId then
            existingJobId =
                existingJobId:
                match("^%s*(.-)%s*$")
        end

        if existingJobId == jobId then

            alreadyExists = true
            break
        end
    end

    if alreadyExists then

        print(
            "[AntiAFK] ℹ️ JobId มีอยู่ใน Cache แล้ว"
        )

        print(
            "[AntiAFK] JobId:",
            jobId
        )

        return false
    end

    -- =====================================================
    -- ADD JOBID
    -- =====================================================

    local output = content

    if output ~= ""
        and not output:match("\n$")
    then
        output = output .. "\n"
    end

    output =
        output
        .. jobId
        .. "\n"

    local success, err =
        pcall(function()

            writefile(
                SERVER_CACHE_FILE,
                output
            )

        end)

    if not success then

        warn(
            "[AntiAFK] ❌ บันทึก Server Cache ไม่สำเร็จ:",
            err
        )

        return false
    end

    -- เพิ่มเข้า Memory ด้วย
    table.insert(
        serverCache,
        jobId
    )

    print("================================")
    print("[AntiAFK] 💾 เพิ่ม JobId ลง Server Cache")
    print("[AntiAFK] JobId:", jobId)
    print("[AntiAFK] อยู่ครบ:", CACHE_SAVE_TIME, "วินาที")
    print("================================")

    return true
end

-- =========================================================
-- GET CACHE COUNT
-- =========================================================

local function getServerCacheCount()

    return #serverCache
end

-- =========================================================
-- GET RANDOM CACHE JOB
-- =========================================================

local function getRandomCachedJob()

    local currentJobId =
        game.JobId

    local candidates = {}

    for index, jobId in ipairs(serverCache) do

        if jobId
            and jobId ~= ""
            and jobId ~= currentJobId
            and not isBlockedJob(jobId)
        then

            table.insert(
                candidates,
                {
                    JobId = jobId,
                    Index = index
                }
            )
        end
    end

    if #candidates == 0 then
        return nil
    end

    local selected =
        candidates[
            math.random(
                1,
                #candidates
            )
        ]

    return selected.JobId,
           selected.Index
end

-- =========================================================
-- SERVER HOP
-- =========================================================

local function serverHop()

    if getgenv().AntiAFKHopping then

        print(
            "[AntiAFK] Server Hop กำลังทำงานอยู่แล้ว"
        )

        return
    end

    getgenv().AntiAFKHopping = true
    getgenv().AntiAFKRunning = false

    print("================================")
    print("[AntiAFK] 🔄 SERVER CACHE HOP")
    print("[AntiAFK] Current JobId:", game.JobId)
    print(
        "[AntiAFK] Blocked Jobs:",
        getBlockedJobCount()
    )
    print(
        "[AntiAFK] Cache:",
        getServerCacheCount(),
        "JobId"
    )
    print("================================")

    -- โหลด Cache ใหม่
    loadServerCache()

    while true do

        local serverJobId, cacheIndex =
            getRandomCachedJob()

        -- =================================================
        -- CACHE หมด
        -- =================================================

        if not serverJobId then

            warn("================================")
            warn(
                "[AntiAFK] ❌ ไม่มี JobId ใน Cache ที่ใช้ได้"
            )
            warn(
                "[AntiAFK] Cache:",
                getServerCacheCount()
            )
            warn(
                "[AntiAFK] Blocked:",
                getBlockedJobCount()
            )
            warn("================================")

            getgenv().AntiAFKHopping = false

            return false
        end

        print("================================")
        print(
            "[AntiAFK] 🎲 สุ่ม JobId จาก Cache"
        )
        print(
            "[AntiAFK] Cache Index:",
            cacheIndex
        )
        print(
            "[AntiAFK] JobId:",
            serverJobId
        )
        print("================================")

        local teleportFailed = false
        local teleportResult = nil
        local teleportMessage = nil

        local connection

        connection =
            TeleportService.TeleportInitFailed:
            Connect(
                function(
                    player,
                    result,
                    message
                )

                    if player ~= Player then
                        return
                    end

                    teleportFailed = true
                    teleportResult = result
                    teleportMessage = message

                end
            )

        -- =================================================
        -- TELEPORT
        -- =================================================

        local callSuccess,
            callError =
            pcall(function()

                TeleportService:
                    TeleportToPlaceInstance(
                        game.PlaceId,
                        serverJobId,
                        Player
                    )

            end)

        if not callSuccess then

            warn("================================")
            warn(
                "[AntiAFK] ❌ Teleport Call Error"
            )
            warn(
                "[AntiAFK] JobId:",
                serverJobId
            )
            warn(
                "[AntiAFK] Error:",
                callError
            )
            warn("================================")

            if connection then
                connection:Disconnect()
            end

            task.wait(
                HOP_RETRY_DELAY
            )

            continue
        end

        -- =================================================
        -- WAIT RESULT
        -- =================================================

        local timeout = 0

        while timeout < 10 do

            task.wait(0.25)

            timeout += 0.25

            if teleportFailed then
                break
            end
        end

        if connection then
            connection:Disconnect()
        end

        -- =================================================
        -- FAILED
        -- =================================================

        if teleportFailed then

            warn("================================")
            warn(
                "[AntiAFK] ❌ Teleport ไม่สำเร็จ"
            )
            warn(
                "[AntiAFK] JobId:",
                serverJobId
            )
            warn(
                "[AntiAFK] Result:",
                tostring(
                    teleportResult
                )
            )
            warn(
                "[AntiAFK] Message:",
                tostring(
                    teleportMessage
                )
            )
            warn(
                "[AntiAFK] 🔄 สุ่ม JobId ใหม่"
            )
            warn("================================")

            task.wait(
                HOP_RETRY_DELAY
            )

            continue
        end

        -- =================================================
        -- SUCCESS
        -- =================================================

        print("================================")
        print(
            "[AntiAFK] 📡 ส่ง Teleport แล้ว"
        )
        print(
            "[AntiAFK] JobId:",
            serverJobId
        )
        print(
            "[AntiAFK] ⏳ รอเข้า Server ใหม่..."
        )
        print("================================")

        return true
    end
end

-- =========================================================
-- POSITION
-- =========================================================

local samePositionCount = 0
local lastPosition = nil

local function getPosition()

    local currentCharacter =
        Player.Character

    if not currentCharacter then
        return nil
    end

    local currentRoot =
        currentCharacter:
        FindFirstChild(
            "HumanoidRootPart"
        )

    if not currentRoot then
        return nil
    end

    return currentRoot.Position
end

-- =========================================================
-- POSITION CHECK
-- =========================================================

local function isSamePosition(
    pos1,
    pos2
)

    if not pos1 or not pos2 then
        return false
    end

    local distance =
        (pos1 - pos2).Magnitude

    print(
        "[AntiAFK] ระยะที่ขยับ:",
        math.floor(
            distance * 100
        ) / 100,
        "studs"
    )

    return distance <=
        POSITION_TOLERANCE
end

    -- =====================================================

            "[AntiAFK] บันทึกตำแหน่งเริ่มต้นแล้ว"

    -- =====================================================

    elseif isSamePosition(
        lastPosition,

    ) then

            "[AntiAFK] อยู่จุดเดิม | "
            .. samePositionCount
            .. "/"
            .. REQUIRED_CHECKS
        )

            REQUIRED_CHECKS
        then

                "[AntiAFK] ⚠️ AFK TRIGGERED"
            )

            local currentJobId =
                game.JobId

            print(
                "[AntiAFK] นิ่งครบ "
                .. REQUIRED_CHECKS
                .. " ครั้ง"
            )

            -- =================================================
            -- SAVE BLOCKED JOB
            -- =================================================

                currentJobId
            )

            print(
                "[AntiAFK] 💾 บันทึก Blocked JobId แล้ว"
            )

            print(
                "[AntiAFK] 🔴 สาเหตุ: AFK"
            )

            print(
                "[AntiAFK] 🔴 กำลังเตะออก..."
            )

            Player:Kick(

            "[AntiAFK] ตำแหน่งเปลี่ยน"
            .. " -> Reset 0/"
            .. REQUIRED_CHECKS
        )
    end

    lastPosition =
        currentPosition
