-- =========================================================
-- LOAD BLOCKED JOB FILE
-- =========================================================

local function loadJobFile()

    if not hasFileAPI then
        return
    end

    if not isfile(JOB_FILE) then
        print("[AntiAFK] ยังไม่มี", JOB_FILE)
        return
    end

    local success, content = pcall(function()
        return readfile(JOB_FILE)
    end)

    if not success then
        warn("[AntiAFK] อ่าน Job File ไม่สำเร็จ:", content)
        return
    end

    if not content or content == "" then
        return
    end

    local count = 0

    for line in string.gmatch(content, "[^\r\n]+") do

        local jobId = line:match("^([^|]+)")

        if jobId then
            jobId = jobId:match("^%s*(.-)%s*$")
        end

        if jobId and jobId ~= "" then

            if not blockedJobs[jobId] then
                blockedJobs[jobId] = true
                count += 1
            end

        end
    end

    print("[AntiAFK] โหลด Blocked JobId:", count)
end

loadJobFile()

-- =========================================================
-- COUNT BLOCKED
-- =========================================================

local function getBlockedJobCount()

    local count = 0

    for _ in pairs(blockedJobs) do
        count += 1
    end

    return count
end

-- =========================================================
-- CHECK BLOCKED
-- =========================================================

local function isBlockedJob(jobId)

    if not jobId or jobId == "" then
        return false
    end

    return blockedJobs[jobId] == true
end

-- =========================================================
-- SAVE BLOCKED JOB FILE
-- =========================================================

local function saveJobFile()

    if not hasFileAPI then
        return false
    end

    local jobs = {}

    for jobId in pairs(blockedJobs) do
        table.insert(jobs, jobId)
    end

    -- จำกัดจำนวน
    while #jobs > MAX_BLOCKED_JOBS do

        local removeId = table.remove(jobs, 1)

        if removeId then
            blockedJobs[removeId] = nil
        end
    end

    local output = ""

    for _, jobId in ipairs(jobs) do
        output = output .. jobId .. "\n"
    end

    local success, err = pcall(function()
        writefile(JOB_FILE, output)
    end)

    if not success then

        warn(
            "[AntiAFK] บันทึก Blocked Job ไม่สำเร็จ:",
            err
        )

        return false
    end

    return true
end

-- =========================================================
-- ADD BLOCKED JOB
-- =========================================================

local function addBlockedJob(jobId)

    if not jobId or jobId == "" then
        return false
    end

    -- กันซ้ำ
    if blockedJobs[jobId] then

        print(
            "[AntiAFK] ⚠️ JobId นี้อยู่ใน Blocked แล้ว:",
            jobId
        )

        return true
    end

    blockedJobs[jobId] = true

    print("================================")
    print("[AntiAFK] 🚫 เพิ่ม Blocked JobId")
    print("[AntiAFK] JobId:", jobId)
    print("================================")

    saveJobFile()

    return true
end

    return count
end

-- =========================================================
-- LOAD SERVER CACHE
-- =========================================================

local function loadServerCache()

    serverCache = {}

    if not hasFileAPI then
        warn("[AntiAFK] File API ไม่มี -> อ่าน Cache ไม่ได้")
        return false
    end

    if not isfile(SERVER_CACHE_FILE) then

        print(
            "[AntiAFK] ยังไม่มี",
            SERVER_CACHE_FILE
        )

        return false
    end

    local success, content = pcall(function()
        return readfile(SERVER_CACHE_FILE)
    end)

    if not success then

        warn(
            "[AntiAFK] อ่าน Server Cache ไม่สำเร็จ:",
            content
        )

        return false
    end

    if not content or content == "" then
        return false
    end

    local duplicateCount = 0
    local blockedCount = 0
    local count = 0

    for line in string.gmatch(content, "[^\r\n]+") do

        local jobId =
            line:match("^([^|]+)")

        local players =
            tonumber(
                line:match("^[^|]+|([^|]+)")
            )

        local maxPlayers =
            tonumber(
                line:match("^[^|]+|[^|]+|([^|]+)")
            )

        if jobId then
            jobId = jobId:match("^%s*(.-)%s*$")
        end

        if jobId and jobId ~= "" then

            -- =============================================
            -- ถ้า Blocked ไม่เอาเข้า Cache
            -- =============================================

            if isBlockedJob(jobId) then

                blockedCount += 1

            -- =============================================
            -- กัน JobId ซ้ำ
            -- =============================================

            elseif serverCache[jobId] then

                duplicateCount += 1

            else

                serverCache[jobId] = {
                    players = players or 0,
                    maxPlayers = maxPlayers or 0
                }

                count += 1
            end
        end
    end

    print("================================")
    print("[AntiAFK] 📦 โหลด Server Cache")
    print("[AntiAFK] Valid:", count)
    print("[AntiAFK] Duplicate:", duplicateCount)
    print("[AntiAFK] Blocked:", blockedCount)
    print("================================")

    return count > 0
end

-- =========================================================
-- SAVE SERVER CACHE
-- =========================================================

local function saveServerCache()

    if not hasFileAPI then
        return false
    end

    local output = ""

    local jobs = {}

    for jobId, data in pairs(serverCache) do

        if not isBlockedJob(jobId) then

            table.insert(jobs, {
                jobId = jobId,
                players = data.players or 0,
                maxPlayers = data.maxPlayers or 0
            })

        end
    end

    -- =====================================================
    -- จำกัด Cache
    -- =====================================================

    while #jobs > MAX_SERVER_CACHE do
        table.remove(jobs, 1)
    end

    -- =====================================================
    -- เขียนไฟล์
    -- =====================================================

    for _, data in ipairs(jobs) do

        output =
            output
            .. data.jobId
            .. "|"
            .. tostring(data.players)
            .. "|"
            .. tostring(data.maxPlayers)
            .. "\n"
    end

    local success, err = pcall(function()

        writefile(
            SERVER_CACHE_FILE,
            output
        )

    end)

    if not success then

        warn(
            "[AntiAFK] ❌ Save Server Cache ไม่สำเร็จ:",
            err
        )

        return false
    end

    print(
        "[AntiAFK] 💾 อัปเดต Server Cache:",
        #jobs,
        "JobId"
    )

    return true
end

-- =========================================================
-- REMOVE BLOCKED FROM CACHE
-- =========================================================

local function removeBlockedFromCache()

    local removed = 0

    for jobId in pairs(serverCache) do

        if isBlockedJob(jobId) then

            serverCache[jobId] = nil
            removed += 1

        end
    end

    if removed > 0 then

        print(
            "[AntiAFK] 🧹 ลบ Blocked JobId จาก Cache:",
            removed
        )

        saveServerCache()
    end
end

-- =========================================================
-- UPDATE CURRENT SERVER CACHE
-- =========================================================

local function updateCurrentServerCache()

    local currentJobId = game.JobId

    if not currentJobId or currentJobId == "" then
        return false
    end

    -- =============================================
    -- ถ้าปัจจุบันเป็น Blocked
    -- ไม่เพิ่ม Cache
    -- =============================================

    if isBlockedJob(currentJobId) then

        print(
            "[AntiAFK] 🚫 Current JobId เป็น Blocked"
        )

        -- ถ้ามีอยู่ใน Cache อยู่แล้ว ให้ลบออก
        if serverCache[currentJobId] then

            serverCache[currentJobId] = nil

            saveServerCache()

        end

        return false
    end

    local currentPlayers =
        #Players:GetPlayers()

    local maxPlayers =
        Players.MaxPlayers

    -- =============================================
    -- Update เดิมแทนการเพิ่มซ้ำ
    -- =============================================

    local alreadyExists =
        serverCache[currentJobId] ~= nil

    serverCache[currentJobId] = {
        players = currentPlayers,
        maxPlayers = maxPlayers
    }

    if alreadyExists then

        print("================================")
        print("[AntiAFK] 🔄 อัปเดต Server เดิม")
        print("[AntiAFK] JobId:", currentJobId)
        print(
            "[AntiAFK] Players:",
            currentPlayers .. "/" .. maxPlayers
        )
        print("================================")

    else

        print("================================")
        print("[AntiAFK] ➕ เพิ่ม Server ใหม่")
        print("[AntiAFK] JobId:", currentJobId)
        print(
            "[AntiAFK] Players:",
            currentPlayers .. "/" .. maxPlayers
        )
        print("================================")
    end

    saveServerCache()

    return true
end

-- =========================================================
-- GET AVAILABLE CACHE JOBS
-- =========================================================

local function getAvailableCachedJobs()

    local currentJobId =
        game.JobId

    local candidates = {}

    for jobId, data in pairs(serverCache) do

        if jobId
            and jobId ~= ""
            and jobId ~= currentJobId
            and not isBlockedJob(jobId)
        then

            local players =
                tonumber(data.players) or 0

            local maxPlayers =
                tonumber(data.maxPlayers) or 0

            -- =============================================
            -- ถ้ามี MaxPlayers
            -- ต้องไม่เต็ม
            -- =============================================

            if maxPlayers > 0 then

                if players < maxPlayers then

                    table.insert(
                        candidates,
                        {
                            jobId = jobId,
                            players = players,
                            maxPlayers = maxPlayers
                        }
                    )

                end

            else

                -- ถ้าไม่มีข้อมูล MaxPlayers
                -- ยังอนุญาตให้ลอง
                table.insert(
                    candidates,
                    {
                        jobId = jobId,
                        players = players,
                        maxPlayers = maxPlayers
                    }
                )

            end
        end
    end

    return candidates
end

-- =========================================================
-- RANDOM SHUFFLE
-- =========================================================

local function shuffle(tbl)

    for i = #tbl, 2, -1 do

        local j =
            math.random(
                1,
                i
            )

        tbl[i], tbl[j] =
            tbl[j], tbl[i]
    end

    return tbl
end

-- =========================================================
-- SERVER HOP
-- =========================================================

local function serverHop()

    if getgenv().AntiAFKHopping then

        print(
            "[AntiAFK] Server Hop กำลังทำงานอยู่แล้ว"
        )

        return false
    end

    getgenv().AntiAFKHopping = true
    getgenv().AntiAFKRunning = false

    print("================================")
    print("[AntiAFK] 🔄 SERVER CACHE HOP")
    print("[AntiAFK] Current:", game.JobId)
    print("[AntiAFK] Blocked:", getBlockedJobCount())
    print("[AntiAFK] Cache:", getServerCacheCount())
    print("================================")

    -- โหลด Cache ล่าสุด
    loadServerCache()

    -- ลบ Blocked ออกจาก Cache
    removeBlockedFromCache()

    -- =============================================
    -- หา Server ที่ยังไม่เต็ม
    -- =============================================

    local candidates =
        getAvailableCachedJobs()

    if #candidates == 0 then

        warn("================================")
        warn("[AntiAFK] ❌ ไม่มี Server ที่ใช้ได้")
        warn(
            "[AntiAFK] Cache:",
            getServerCacheCount()
        )
        warn(
            "[AntiAFK] Blocked:",
            getBlockedJobCount()
        )
        warn("================================")

        getgenv().AntiAFKHopping = false

        return false
    end

    -- =============================================
    -- สุ่ม JobId
    -- =============================================

    shuffle(candidates)

    print(
        "[AntiAFK] 🎲 Server ที่เลือกได้:",
        #candidates
    )

    -- =============================================
    -- ลองทีละตัว
    -- =============================================

    for index, serverData in ipairs(candidates) do

        local serverJobId =
            serverData.jobId

        print("================================")
        print(
            "[AntiAFK] 🎯 ลอง Server:",
            index .. "/" .. #candidates
        )
        print(
            "[AntiAFK] JobId:",
            serverJobId
        )
        print(
            "[AntiAFK] Players:",
            serverData.players
            .. "/"
            .. serverData.maxPlayers
        )
        print("================================")

        local teleportFailed = false
        local teleportResult = nil
        local teleportMessage = nil

        local connection

        connection =
            TeleportService.TeleportInitFailed:
            Connect(
                function(
                    player,
                    result,
                    message
                )

                    if player ~= Player then
                        return
                    end

                    teleportFailed = true
                    teleportResult = result
                    teleportMessage = message

                end
            )

        local callSuccess,
            callError =
            pcall(function()

                TeleportService:
                    TeleportToPlaceInstance(
                        game.PlaceId,
                        serverJobId,
                        Player
                    )

            end)

        if not callSuccess then

            warn(
                "[AntiAFK] ❌ Teleport Error:",
                callError
            )

            if connection then
                connection:Disconnect()
            end

            task.wait(
                HOP_RETRY_DELAY
            )

            continue
        end

        -- =============================================
        -- รอผล
        -- =============================================

        local timeout = 0

        while timeout < TELEPORT_TIMEOUT do

            task.wait(0.25)

            timeout += 0.25

            if teleportFailed then
                break
            end
        end

        if connection then
            connection:Disconnect()
        end

        if teleportFailed then

            warn("================================")
            warn("[AntiAFK] ❌ เข้า Server ไม่สำเร็จ")
            warn("[AntiAFK] JobId:", serverJobId)
            warn(
                "[AntiAFK] Result:",
                tostring(teleportResult)
            )
            warn(
                "[AntiAFK] Message:",
                tostring(teleportMessage)
            )
            warn("[AntiAFK] 🔄 ลองตัวถัดไป")
            warn("================================")

            task.wait(
                HOP_RETRY_DELAY
            )

            continue
        end

        print("================================")
        print("[AntiAFK] 📡 ส่ง Teleport แล้ว")
        print("[AntiAFK] JobId:", serverJobId)
        print("[AntiAFK] ⏳ รอเข้า Server...")
        print("================================")

        return true
    end

    warn(
        "[AntiAFK] ❌ ลองทุก Server แล้วแต่เข้าไม่ได้"
    )

    getgenv().AntiAFKHopping = false

    return false
end

local samePositionCount = 0
local lastPosition = nil

local function getPosition()

    local currentCharacter =
        Player.Character

    if not currentCharacter then
        return nil
    end

    local currentRoot =
        currentCharacter:
        FindFirstChild(
            "HumanoidRootPart"
        )

    if not currentRoot then
        return nil
    end

    return currentRoot.Position
end

-- =========================================================
-- POSITION CHECK
-- =========================================================

local function isSamePosition(
    pos1,
    pos2
)

    if not pos1 or not pos2 then
        return false
    end

    local distance =
        (pos1 - pos2).Magnitude

    print(
        "[AntiAFK] ระยะที่ขยับ:",
        math.floor(
            distance * 100
        ) / 100,
        "studs"
    )

    return distance <=
        POSITION_TOLERANCE
end
