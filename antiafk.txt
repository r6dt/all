-- =========================================================
-- LOAD BLOCKED JOB FILE
-- =========================================================

local function loadJobFile()

    if not hasFileAPI then
        return
    end

    if not isfile(JOB_FILE) then

        print(
            "[AntiAFK] ยังไม่มี",
            JOB_FILE
        )

        return
    end

    local success, content =
        pcall(function()
            return readfile(JOB_FILE)
        end)

    if not success then

        warn(
            "[AntiAFK] อ่าน Job File ไม่สำเร็จ:",
            content
        )

        return
    end

    if not content or content == "" then
        return
    end

    local count = 0

    for line in string.gmatch(
        content,
        "[^\r\n]+"
    ) do

        local jobId =
            line:match("^([^|]+)")

        if jobId then

            jobId =
                jobId:match(
                    "^%s*(.-)%s*$"
                )

        end

        if jobId
            and jobId ~= ""
        then

            if not blockedJobs[jobId] then

                blockedJobs[jobId] = true
                count += 1

            end
        end
    end

    print(
        "[AntiAFK] โหลด Blocked JobId:",
        count
    )
end

loadJobFile()

-- =========================================================
-- CHECK BLOCKED JOB
-- =========================================================

local function isBlockedJob(jobId)

    if not jobId
        or jobId == ""
    then
        return false
    end

    return blockedJobs[jobId] == true
end

-- =========================================================
-- COUNT BLOCKED
-- =========================================================

local function getBlockedJobCount()

    local count = 0

    for _ in pairs(blockedJobs) do
        count += 1
    end

    return count
end

-- =========================================================
-- SAVE BLOCKED JOB FILE
-- =========================================================

local function saveJobFile()

    if not hasFileAPI then
        return false
    end

    local jobs = {}

    for jobId in pairs(blockedJobs) do

        table.insert(
            jobs,
            jobId
        )

    end

    -- =====================================================
    -- จำกัดจำนวน Blocked สูงสุด
    -- =====================================================

    while #jobs > MAX_BLOCKED_JOBS do

        local removeId =
            table.remove(
                jobs,
                1
            )

        if removeId then
            blockedJobs[removeId] = nil
        end
    end

    local output = ""

    for _, jobId in ipairs(jobs) do

        output =
            output
            .. jobId
            .. "\n"

    end

    local success, err =
        pcall(function()

            writefile(
                JOB_FILE,
                output
            )

        end)

    if not success then

        warn(
            "[AntiAFK] บันทึก Blocked Job ไม่สำเร็จ:",
            err
        )

        return false
    end

    return true
end

-- =========================================================
-- ADD BLOCKED JOB
-- =========================================================

local function addBlockedJob(jobId)

    if not jobId
        or jobId == ""
    then
        return false
    end

    if blockedJobs[jobId] then

        print(
            "[AntiAFK] ⚠️ JobId นี้ Blocked อยู่แล้ว:",
            jobId
        )

        return true
    end

    blockedJobs[jobId] = true

    print("================================")
    print("[AntiAFK] 🚫 เพิ่ม Blocked JobId")
    print("[AntiAFK] JobId:", jobId)
    print("================================")

    saveJobFile()

    return true
end

-- =========================================================
-- GET LIVE SERVERS
-- =========================================================
--
-- HTTP จะทำงานเฉพาะตอน serverHop() เรียกฟังก์ชันนี้
--
-- ไม่ทำงานตอน:
-- - เข้าเกม
-- - อยู่เซิร์ฟเวอร์ปกติ
-- - Anti-AFK Check
-- - Status
--
-- =========================================================

local function getLiveServers()

    local servers = {}

    local seenServers = {}

    local cursor = nil

    print("================================")
    print("[AntiAFK] 🌐 กำลังโหลด Live Server List")
    print("[AntiAFK] HTTP: START")
    print("================================")

    for page = 1, LIVE_SERVER_MAX_PAGES do

        local url =
            "https://games.roblox.com/v1/games/"
            .. tostring(game.PlaceId)
            .. "/servers/Public?sortOrder=Asc&limit="
            .. tostring(LIVE_SERVER_LIMIT)

        if cursor
            and cursor ~= ""
        then

            url =
                url
                .. "&cursor="
                .. game:GetService("HttpService"):UrlEncode(
                    cursor
                )

        end

        local requestSuccess,
            response =
            pcall(function()

                return game:HttpGet(
                    url
                )

            end)

        if not requestSuccess then

            warn(
                "[AntiAFK] ❌ HTTP Error:"
            )

            warn(
                "[AntiAFK] Page:",
                page
            )

            warn(
                "[AntiAFK] Error:",
                response
            )

            break
        end

        local decodeSuccess,
            data =
            pcall(function()

                return game:GetService("HttpService"):
                    JSONDecode(
                        response
                    )

            end)

        if not decodeSuccess then

            warn(
                "[AntiAFK] ❌ JSON Decode Error:"
            )

            warn(
                "[AntiAFK] Page:",
                page
            )

            warn(
                "[AntiAFK] Error:",
                data
            )

            break
        end

        if type(data) ~= "table" then

            warn(
                "[AntiAFK] ❌ Server List ไม่ใช่ Table"
            )

            break
        end

        if type(data.data) ~= "table" then

            warn(
                "[AntiAFK] ❌ ไม่พบ Server Data"
            )

            break
        end

        -- =================================================
        -- อ่าน Server
        -- =================================================

        for _, server in ipairs(
            data.data
        ) do

            if type(server) == "table" then

                local serverId =
                    server.id

                local playing =
                    tonumber(
                        server.playing
                    ) or 0

                local maxPlayers =
                    tonumber(
                        server.maxPlayers
                    ) or 0

                if serverId
                    and serverId ~= ""
                then

                    -- =============================================
                    -- Current JobId
                    -- =============================================

                    if serverId ~= game.JobId then

                        -- =========================================
                        -- Blocked JobId
                        -- =========================================

                        if not isBlockedJob(
                            serverId
                        ) then

                            -- =====================================
                            -- ต้องไม่เต็ม
                            -- =====================================

                            if maxPlayers <= 0
                                or playing < maxPlayers
                            then

                                if not seenServers[
                                    serverId
                                ] then

                                    seenServers[
                                        serverId
                                    ] = true

                                    table.insert(
                                        servers,
                                        {

                                            jobId =
                                                serverId,

                                            players =
                                                playing,

                                            maxPlayers =
                                                maxPlayers
                                        }
                                    )

                                end
                            end
                        end
                    end
                end
            end
        end

        -- =================================================
        -- Pagination
        -- =================================================

        cursor =
            data.nextPageCursor

        if not cursor
            or cursor == ""
        then

            break
        end
    end

    print("================================")
    print(
        "[AntiAFK] 🌐 Live Server พบ:",
        #servers
    )
    print(
        "[AntiAFK] HTTP: END"
    )
    print("================================")

    return servers
end

-- =========================================================
-- SORT LIVE SERVERS
-- =========================================================
--
-- คนน้อย -> คนเยอะ
--
-- ถ้าคนเท่ากัน:
-- ไม่ได้สุ่ม
-- ใช้ลำดับที่ API ส่งกลับมา
--
-- =========================================================

local function sortLowPlayerFirst(
    servers
)

    table.sort(
        servers,
        function(a, b)

            if a.players ~= b.players then

                return a.players < b.players

            end

            return a.jobId < b.jobId
        end
    )

    return servers
end

local function serverHop()

    -- HTTP LIVE SERVER LIST

    local validServers =
        getLiveServers()

    if #validServers == 0 then

        warn("================================")
        warn(
            "[AntiAFK] ❌ ไม่มี Live Server ที่ใช้ได้"
        )
        warn(
            "[AntiAFK] Blocked:",

            false

        return false
    end

    -- =====================================================
    -- LOW PLAYER FIRST
    -- =====================================================

    sortLowPlayerFirst(
        validServers
    )

    print("================================")
    print(
        "[AntiAFK] 📊 Server เรียงจากคนน้อย -> คนเยอะ"
    )
    print(
        "[AntiAFK] จำนวน Server:",
        #validServers
    )
    print("================================")

    -- =====================================================
    -- แสดงลำดับ Server
    -- =====================================================

    for index, serverInfo in ipairs(
        validServers
    ) do

        print(
            "[AntiAFK] #"
            .. index
            .. " | Players: "
            .. serverInfo.players
            .. "/"
            .. serverInfo.maxPlayers
            .. " | JobId: "
            .. serverInfo.jobId
        )

    end

    print("================================")

    -- =====================================================
    -- ลองทีละ JobId
    -- =====================================================

    for index, serverInfo in ipairs(
        validServers
    ) do

        local serverJobId =
            serverInfo.jobId

        -- =================================================
        -- ตรวจซ้ำก่อน Teleport
        -- =================================================

        if serverJobId == game.JobId then

            print(
                "[AntiAFK] ⚠️ ข้าม Current JobId"
            )

            continue
        end

        if isBlockedJob(
            serverJobId
        ) then

            print(
                "[AntiAFK] 🚫 ข้าม Blocked JobId:",
                serverJobId
            )

            continue
        end

        print("================================")

        print(
            "[AntiAFK] 🎯 Low Player Server:",
            index
            .. "/"
            .. #validServers
        )

        print(
            "[AntiAFK] JobId:",
            serverJobId
        )

        print(
            "[AntiAFK] Players:",
            serverInfo.players
            .. "/"
            .. serverInfo.maxPlayers
        )

        print("================================")

        local teleportFailed =
            false

        local teleportResult =
            nil

        local teleportMessage =
            nil

        local connection

        connection =
            TeleportService
            .TeleportInitFailed
            :Connect(
                function(
                    player,
                    result,
                    message
                )

                    if player ~= Player then
                        return
                    end

                    teleportFailed =
                        true

                    teleportResult =
                        result

                    teleportMessage =
                        message
                end
            )

        -- =================================================
        -- TELEPORT
        -- =================================================

        local callSuccess,
            callError =
            pcall(function()

                TeleportService:
                    TeleportToPlaceInstance(
                        game.PlaceId,
                        serverJobId,
                        Player
                    )

            end)

        if not callSuccess then

            warn(
                "[AntiAFK] ❌ Teleport Error:",
                callError
            )

            if connection then
                connection:Disconnect()
            end

            print(
                "[AntiAFK] ⏳ รอ",
                HOP_RETRY_DELAY,
                "วินาที ก่อนลอง Server ถัดไป"
            )

            task.wait(
                HOP_RETRY_DELAY
            )

            continue
        end

        -- =================================================
        -- WAIT TELEPORT RESULT
        -- =================================================

        local timeout =
            0

        while timeout <
            TELEPORT_TIMEOUT
        do

            task.wait(0.25)

            timeout += 0.25

            if teleportFailed then
                break
            end
        end

        if connection then
            connection:Disconnect()
        end

        -- =================================================
        -- TELEPORT FAILED
        -- =================================================

        if teleportFailed then

            warn("================================")

            warn(
                "[AntiAFK] ❌ เข้า Server ไม่สำเร็จ"
            )

            warn(
                "[AntiAFK] JobId:",
                serverJobId
            )

            warn(
                "[AntiAFK] Players:",
                serverInfo.players
                .. "/"
                .. serverInfo.maxPlayers
            )

            warn(
                "[AntiAFK] Result:",
                tostring(
                    teleportResult
                )
            )

            warn(
                "[AntiAFK] Message:",
                tostring(
                    teleportMessage
                )
            )

            warn(
                "[AntiAFK] 🔄 ลอง Server ถัดไป"
            )

            warn("================================")

            print(
                "[AntiAFK] ⏳ รอ",
                HOP_RETRY_DELAY,
                "วินาที"
            )

            task.wait(
                HOP_RETRY_DELAY
            )

            continue
        end

        -- =================================================
        -- TELEPORT REQUEST SENT
        -- =================================================

        print("================================")

        print(
            "[AntiAFK] 📡 ส่ง Teleport แล้ว"
        )

        print(
            "[AntiAFK] JobId:",
            serverJobId
        )

        print(
            "[AntiAFK] Players:",
            serverInfo.players
            .. "/"
            .. serverInfo.maxPlayers
        )

        print(
            "[AntiAFK] ⏳ รอเข้า Server..."
        )

        print("================================")

        return true
    end

    -- =====================================================
    -- ทุก Server ล้มเหลว
    -- =====================================================

    warn(
        "[AntiAFK] ❌ ไม่มี Server ที่ Teleport สำเร็จ"
    )

    getgenv().AntiAFKHopping =
        false

    return false
end

-- =========================================================
-- GET POSITION
-- =========================================================

local function getPosition()

    local currentCharacter =
        Player.Character

    if not currentCharacter then
        return nil
    end

    local currentRoot =
        currentCharacter:
        FindFirstChild(
            "HumanoidRootPart"
        )

    if not currentRoot then
        return nil
    end

    return currentRoot.Position
end

-- =========================================================
-- CHECK SAME POSITION
-- =========================================================

local function isSamePosition(
    pos1,
    pos2
)

    if not pos1
        or not pos2
    then

        return false
    end

    local distance =
        (pos1 - pos2).Magnitude

    print(
        "[AntiAFK] ระยะที่ขยับ:",
        math.floor(
            distance * 100
        ) / 100,
        "studs"
    )

    return distance <=
        POSITION_TOLERANCE
end
